How it plays

The squares on the board are graded by size and called pieces; the top draws what this level must hand in, and the field has rings. Click a place on the board: if there is only one piece within range, it jumps to the click; if there are two pieces of the same level, they merge into one piece one level higher, landing at the click. Then the hunters, diamonds and crabs on the field move in small steps, each small step toward the piece nearest to it at that moment; whichever piece it presses on, it bites down one level and flings far away, and the smallest piece is gone when bitten. The new piece just merged in this move cannot be bitten. When everything to be handed in lands whole inside the rings, the level is cleared; in some levels what must be handed in also includes diamonds or crabs. The bottom row is the budget, which shortens at every click on the board, and the game ends when it is deducted to zero.

Model

The final model: 5 object types and 16 rules. Each rule is a Python check against every recorded step; 12 of them are translated into Lean (the paper's per-game table). Code under a rule is the file of the same name in the agent's Lean project.

def reach2 (p : Piece) (r c : Int) : Int :=
  let dr := max (max (p.row - r) 0) (r - (p.row + p.sideI - 1))
  let dc := max (max (p.col - c) 0) (c - (p.col + p.sideI - 1))
  dr * dr + dc * dc

def inReach (ps : List Piece) (r c : Int) : List Piece :=
  ps.filter (fun p => decide (reach2 p r c ≤ 64))

def placedRow (s r : Int) : Int := min (max (r - s / 2) 10) (63 - s + 1)
def placedCol (s c : Int) : Int := min (max (c - s / 2) 0) (64 - s)

def placedPiece (t : Nat) (r c : Int) : Piece :=
  ⟨t, placedRow (side t : Int) r, placedCol (side t : Int) c⟩

structure Resolved where
  pieces : List Piece
  fresh  : Option Piece
deriving DecidableEq, Repr

def resolve (ps : List Piece) (r c : Int) : Option Resolved :=
  match inReach ps r c with
  | [] => some ⟨ps, none⟩
  | [p] =>
      let q := placedPiece p.tier r c
      some ⟨ps.map (fun z => if z = p then q else z), some q⟩
  | [a, b] =>
      if a.tier = b.tier ∧ a.tier < 6 then
        let q := placedPiece (a.tier + 1) r c
        some ⟨(ps.filter (fun z => z ≠ a ∧ z ≠ b)) ++ [q], some q⟩
      else some ⟨ps, none⟩
  | _ => none

def ident (ps : List Piece) (r c : Int) (res : Resolved) (p : Piece) : Piece :=
  if p ∈ inReach ps r c then res.fresh.getD p else p

def cCentre (c : Creature) : Int × Int := (c.row * 104 + rowOff c.kind, c.col * 104 + 208)
def pCentre (p : Piece) : Int × Int := (p.row * 104 + (p.sideI - 1) * 52, p.col * 104 + (p.sideI - 1) * 52)

def dist2 (a b : Int × Int) : Int := (a.1 - b.1) * (a.1 - b.1) + (a.2 - b.2) * (a.2 - b.2)

def sgn (d : Int) : Int := if 52 ≤ d then 1 else if d < -52 then -1 else 0

def touches (c : Creature) (p : Piece) : Bool :=
  decide (p.row < c.row + 4) && decide (c.row < p.row + p.sideI) &&
  decide (p.col < c.col + 5) && decide (c.col < p.col + p.sideI)

def place (k : Nat) (r c : Int) : Creature := ⟨k, min (max r 10) 60, min (max c 0) 59⟩

def clampI (x lo hi : Int) : Int := max lo (min hi x)

def slideAt (a b : Int × Int) (i : Nat) : Int × Int :=
  (a.1 + clampI (b.1 - a.1) (-(416 * (i : Int))) (416 * (i : Int)),
   a.2 + clampI (b.2 - a.2) (-(416 * (i : Int))) (416 * (i : Int)))

structure Cand where
  src     : Piece
  dest    : Piece
  centre  : Int × Int
  arrived : Bool
deriving DecidableEq, Repr

def Cand.at (k : Cand) : Piece :=
  ⟨k.dest.tier, (k.centre.1 - (k.dest.sideI - 1) * 52) / 104, (k.centre.2 - (k.dest.sideI - 1) * 52) / 104⟩

def cands (srcs : List Piece) (dst : Piece → Piece) (i : Nat) : List Cand :=
  srcs.map (fun p =>
    let x := slideAt (pCentre p) (pCentre (dst p)) i
    ⟨p, dst p, x, x == pCentre (dst p)⟩)

def nearest (cc : Int × Int) : List Cand → Option Cand
  | [] => none
  | k :: ks => some (ks.foldl (fun best x => if dist2 cc x.centre < dist2 cc best.centre then x else best) k)

def pickPoint (c : Creature) : Int × Int := (c.row * 104 + 208, c.col * 104 + 208)

def Cand.still (k : Cand) : Bool := k.src == k.dest

def choose (prev : Option Cand) (cs : List Cand) (best : Cand) : Cand :=
  match prev with
  | some p => if p.still && best.still then (cs.find? (fun k => k.src == p.src)).getD best else best
  | none => best

def speed : Nat → Int
  | 13 => 2
  | _ => 1

def chase (prev : Option Cand) (c : Creature) (cs : List Cand) : Option (Creature × Cand × Bool) :=
  match nearest (pickPoint c) cs with
  | none => none
  | some b =>
      let t := choose prev cs b
      let sr := sgn (t.centre.1 - (cCentre c).1)
      let sc := sgn (t.centre.2 - (cCentre c).2)
      if sr = 0 ∧ sc = 0 then some (c, t, true)
      else some (place c.kind (c.row + speed c.kind * sr) (c.col + speed c.kind * sc), t, false)

def chaseRun (srcs : List Piece) (dst : Piece → Piece) : Nat → Nat → Option Cand → Creature → Creature
  | 0, _, _, c => c
  | n + 1, i, prev, c =>
      match chase prev c (cands srcs dst i) with
      | none => c
      | some (c', t, stop) => if stop then c' else chaseRun srcs dst n (i + 1) (some t) c'

def nearClick (c : Creature) (r x : Int) : Bool :=
  let dr := max (max (c.row - r) 0) (r - (c.row + 3))
  let dc := max (max (c.col - x) 0) (x - (c.col + 4))
  decide (dr * dr + dc * dc ≤ 64)

def roundDash (d l2 : Int) : Int :=
  match (List.range 9).find? (fun (i : Nat) =>
      leMulSqrt (5 * (2 * ((i : Int) - 4) - 1)) l2 (34 * d) &&
        !(leMulSqrt (5 * (2 * ((i : Int) - 4) + 1)) l2 (34 * d))) with
  | some i => (i : Int) - 4
  | none => 0

def dashV (c : Creature) (r x : Int) : Int × Int :=
  let dy := r * 104 - (pickPoint c).1
  let dx := x * 104 - (pickPoint c).2
  let l2 := dy * dy + dx * dx
  if l2 = 0 then (0, 0) else (roundDash dy l2, roundDash dx l2)

def pounceV (c : Creature) (r x : Int) : Int × Int :=
  ((r * 104 - (cCentre c).1 + 52) / 104, (x * 104 - (cCentre c).2 + 52) / 104)

def leMulSqrt (a l2 b : Int) : Bool :=
  if a ≤ 0 ∧ 0 ≤ b then true
  else if 0 < a ∧ b < 0 then false
  else if 0 ≤ a then decide (a * a * l2 ≤ b * b)
  else decide (b * b ≤ a * a * l2)

def roundRatio (v l2 : Int) : Int :=
  match (List.range 21).find? (fun (i : Nat) =>
      leMulSqrt (2 * ((i : Int) - 10) - 1) l2 (20 * v) && !(leMulSqrt (2 * ((i : Int) - 10) + 1) l2 (20 * v))) with
  | some i => (i : Int) - 10
  | none => 0

def bitten (p : Piece) (c : Creature) : Option Piece :=
  if p.tier ≤ 1 then none
  else
    let dy := (p.row + p.sideI / 2) - (c.row + 2)
    let dx := (p.col + p.sideI / 2) - (c.col + 2)
    let l2 := dy * dy + dx * dx
    let sr := if l2 = 0 then -10 else roundRatio dy l2
    let sc := if l2 = 0 then 0 else roundRatio dx l2
    let s : Int := (side (p.tier - 1) : Int)
    some ⟨p.tier - 1, min (max (p.row + sr) 10) (63 - s + 1), min (max (p.col + sc) 0) (64 - s)⟩

structure Thing where
  isPiece : Bool
  grade   : Nat
  inRing  : Bool

def pieceCells (p : Piece) : List (Int × Int) :=
  (List.range (side p.tier)).flatMap (fun (i : Nat) =>
    (List.range (side p.tier)).map (fun (j : Nat) => (p.row + (i : Int), p.col + (j : Int))))

def insideSome (cells : List (Int × Int)) (rs : List Ring) : Bool :=
  rs.any (fun r => cells.all (fun x => r.boxHas x.1 x.2))

def centreInside (c : Creature) (rs : List Ring) : Bool :=
  rs.any (fun r =>
    decide (r.row * 104 ≤ (cCentre c).1) && decide ((cCentre c).1 ≤ (r.row + 8) * 104) &&
    decide (r.col * 104 ≤ (cCentre c).2) && decide ((cCentre c).2 ≤ (r.col + 8) * 104))

def things (rs : List Ring) (ps : List Piece) (cs : List Creature) : List Thing :=
  ps.map (fun p => ⟨true, p.tier, insideSome (pieceCells p) rs⟩) ++
  cs.map (fun c => ⟨false, c.kind, centreInside c rs⟩)

def picks {α : Type} : List α → List (α × List α)
  | [] => []
  | x :: xs => (x, xs) :: (picks xs).map (fun p => (p.1, x :: p.2))

def matchAll : List Sample → List Thing → Bool
  | [], _ => true
  | s :: ss, ts =>
      (picks ts).any (fun p =>
        p.1.isPiece == s.isPiece && p.1.grade == s.grade && p.1.inRing && matchAll ss p.2)

def covered (b : Board) (ps : List Piece) (cs : List Creature) : Bool :=
  !b.samples.isEmpty && matchAll b.samples (things b.rings ps cs)

    intro q hq heq
    subst heq
    simp [inReach, List.mem_filter] at hq
    omega
  unfold resolve at h
  split at h
  · simp only [Option.some.injEq] at h
    subst h

What a pass establishes

consistent: for every recorded transition, as read by the agent's parser and with duplicates removed, every rule whose condition holds also has its effect hold. faithful: the certified run, recomputed step by step from the first screen of each level, matches the record and ends in a win. Axioms used: propext, all among Lean's standard ones.

What it does not establish: that the parser reads the screens correctly (that step runs in Python, outside Lean), or that a rule whose condition never occurs in the record is right. The Python file that produced the readings inside the model file is in the archive at meta/repair-20260928/lib/su15-20260924/_lib.py.
