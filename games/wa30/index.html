Replay

The certified run that cleared every level, one action at a time.

Press Play or step to load the run. Keys, while the player is on screen: ← → step, Space play / pause. A click is circled; cells that changed are outlined.

How it plays

The green square is the player; the white edge on top shows which way it faces. Every doughnut has to be brought into a blue-framed container. With empty hands, an arrow key turns the player and, where the way is open, moves it one cell; next to a piece of cargo, the action key grabs it, after which the arrow keys move the player and the cargo together, and pressing the action key again lets go. Cargo can enter fenced areas; the player cannot. After every step each mole on the field moves once: the orange helpers carry cargo that is not in a container into one, and the purple thieves carry cargo back to their own nest, even cargo the player is holding; pressing the action key while facing a thief knocks it out. The level is won when every doughnut sits in a container, the player holds nothing and no mole is carrying anything. The pink timer bar along the bottom shortens with each step; when the steps run out before the level is won, the game is over.

Model

The final model: 8 object types and 8 rules. Each rule is a Python check against every recorded step; 7 of them are translated into Lean (the paper's per-game table). Code under a rule is the file of the same name in the agent's Lean project.

def dragOk (b : Board) (s : St) (f t h : Cell) : Bool :=
  (t == f || s.freeFor b t) &&
  (h == s.player || (b.carryable h && !s.donuts.contains h && !(s.moles.any (fun m => m.pos == h))))

def drag (b : Board) (s : St) (d : Dir) : St :=
  match s.front, s.player.go d with
  | some f, some t =>
    match f.go d with
    | some h =>
      if dragOk b s f t h then
        { s with player := t, donuts := s.donuts.map (fun c => if c == f then h else c) }
      else s
    | none => s
  | _, _ => s

namespace Game

def nth (ms : List Mole) : Nat → Option Mole
  | i => (ms.drop i).head?

def posIn : List Cell → Cell → Nat
  | [], _ => 0
  | y :: ys, x => if y == x then 0 else posIn ys x + 1

def pickLast (ds : List Cell) : List Cell → Option Cell
  | [] => none
  | x :: xs => match pickLast ds xs with
    | none => some x
    | some y => if posIn ds x ≥ posIn ds y then some x else some y

def othersMask (s : St) (m : Mole) : Nat :=
  maskOf (s.moles.map (·.pos)) ^^^ (1 <<< m.pos.idx)

def carryTurn (b : Board) (s : St) (i : Nat) (m : Mole) (o : Dir) : St :=
  match m.pos.go o with
  | none => s
  | some cc =>
    if isHome b m.role cc then
      { s with moles := s.moles.set i { m with cargo := none }, donuts := s.donuts.erase cc ++ [cc] }
    else
      let bl0 := maskOf s.donuts ||| othersMask s m ||| (1 <<< s.player.idx)
      let blocked := if bl0.testBit cc.idx then bl0 ^^^ (1 <<< cc.idx) else bl0
      let ok := fun (v : Cell) => b.walkable v && !v.inMask blocked &&
        (match v.go o with
         | some c2 => b.carryable c2 && (!c2.inMask blocked || c2 == m.pos)
         | none => false)
      let goal := fun (v : Cell) => match v.go o with
        | some c2 => isHome b m.role c2 && !s.donuts.contains c2
        | none => false
      match firstStep ok goal m.pos with
      | none => s
      | some nx =>
        match nx.go o with
        | none => s
        | some nc => { s with moles := s.moles.set i { m with pos := nx },
                              donuts := s.donuts.map (fun c => if c == cc then nc else c) }

def seekTurn (b : Board) (s : St) (i : Nat) (m : Mole) : St × Bool :=
  let adj := Dir.order.filterMap (fun d => match m.pos.go d with
    | some c => if isTarget b s m c then some c else none
    | none => none)
  match pickLast s.donuts adj with
  | some c =>
    let ms := s.moles.map (fun mm => if mm.load == some c then { mm with cargo := none } else mm)
    let hold := if s.holding && s.front == some c then false else s.holding
    ({ s with moles := ms.set i { m with cargo := Dir.between m.pos c }, holding := hold }, true)
  | none =>
    let tmask := s.donuts.foldl (fun acc c => if isTarget b s m c then acc ||| (1 <<< c.idx) else acc) 0
    if tmask == 0 then (s, false) else
    let blocked := maskOf s.donuts ||| othersMask s m ||| (1 <<< s.player.idx)
    let ok := fun (v : Cell) => b.walkable v && !v.inMask blocked
    let goal := fun (v : Cell) => Dir.order.any (fun d => match v.go d with
      | some c => c.inMask tmask
      | none => false)
    match firstStep ok goal m.pos with
    | none => (s, false)
    | some nx => ({ s with moles := s.moles.set i { m with pos := nx } }, false)

def turnOne (b : Board) (acc : St × Bool × Bool) (i : Nat) : St × Bool × Bool :=
  match nth acc.1.moles i with
  | none => acc
  | some m =>
    if (m.role == .helper && acc.2.1) || (m.role == .thief && acc.2.2) then acc else
    match m.cargo with
    | some o => (carryTurn b acc.1 i m o, acc.2.1, acc.2.2)
    | none =>
      let r := seekTurn b acc.1 i m
      if r.2 then (r.1, acc.2.1 || m.role == .helper, acc.2.2 || m.role == .thief)
      else (r.1, acc.2.1, acc.2.2)

def molesAct (b : Board) (s : St) : List Cell × List Mole × Bool :=
  let idx := List.range s.moles.length
  let order := idx.filter (fun i => (nth s.moles i).map (·.role) == some .helper) ++
               idx.filter (fun i => (nth s.moles i).map (·.role) == some .thief)
  let r := order.foldl (turnOne b) (s, false, false)
  (r.1.donuts, r.1.moles, r.1.holding)

end Game

Each theorem is closed by the Lean kernel and is a statement about the learned model.

What a pass establishes

consistent: for every recorded transition, as read by the agent's parser and with duplicates removed, every rule whose condition holds also has its effect hold. faithful: the certified run, recomputed step by step from the first screen of each level, matches the record and ends in a win. Axioms used: propext, all among Lean's standard ones.

What it does not establish: that the parser reads the screens correctly (that step runs in Python, outside Lean), or that a rule whose condition never occurs in the record is right. The Python file that produced the readings inside the model file is in the archive at meta/repair-20260928/lib/wa30-20260920/_lib.py.
