How it plays

On the field there are shelled blocks, yellow cross blocks without a shell, frames, walls, purple bands and door bands. Click a shelled block and control switches to it. A direction key makes it walk that way: if the cell ahead is empty it moves one cell, and if ahead is a wall or a purple band it does not move; when ahead it is adjacent to another block or door band, it does not move itself but pushes the adjacent thing out a fixed distance, stopping on hitting a wall, and continuing to slide if it stops on a band, until it leaves the band. Cross blocks cannot be clicked and can only be pushed. A door band grows one layer with each direction-key press, and when it reaches one layer short of full it resets to zero, and on the beat of the reset it sweeps the block directly in front of it out in the direction it grows. When every block fits exactly into one frame, the level is cleared. The budget bar at the bottom shortens bit by bit with the number of actions, and running it empty is judged a loss.

Model

The final model: 9 object types and 11 rules. Each rule is a Python check against every recorded step; 10 of them are translated into Lean (the paper's per-game table). Code under a rule is the file of the same name in the agent's Lean project.

def clickCell (b : Board) (x y : Nat) : Cell := ⟨(y + 6 - b.pr) / 3, (x + 6 - b.pc) / 3⟩

def pieceAt (s : St) (x : Cell) : Option Nat :=
  (List.range s.pieces.length).find? (fun i => (s.pieces.getD i ⟨[], false, false⟩).cells.contains x)

def clickSelect (b : Board) (s : St) (x y : Nat) : St :=
  match pieceAt s (clickCell b x y) with
  | some i =>
      if (s.pieces.getD i ⟨[], false, false⟩).shelled && i != s.sel then
        { s with
          sel := i
          pieces := ((List.range s.pieces.length).zip s.pieces).map (fun (jp : Nat × Piece) =>
            if jp.1 == s.sel then { jp.2 with retired := true }
            else if jp.1 == i then { jp.2 with retired := false }
            else jp.2) }
      else s
  | none => s

def KICK : Nat := 5

def touched (s : St) (d : Dir) : List Nat :=
  let strip := front s.mine d
  (List.range s.bodies.length).filter (fun i => i != s.sel && overlaps (s.bodies.getD i []) strip)

def pushShove (b : Board) (s : St) (d : Dir) : List (List Cell) :=
  let t := touched s d
  let pos := t.foldl (fun p i => p.set i (slide b d KICK (p.getD i []))) s.bodies
  cascade b d s.sel pos t

def STRIKE : Nat := 3
def REACH : Nat := 3

def strikeOne (b : Board) (np j : Nat) (g : Gate) (pos : List (List Cell)) : List (List Cell) :=
  if g.wraps then
    match g.growth, pos.getD (np + j) [] with
    | some D, top :: _ =>
        let hit := (List.range np).filter (fun i => inFront b top g.h g.w D (pos.getD i []) REACH)
        if hit.isEmpty then pos
        else cascade b D (np + j) (hit.foldl (fun p i => p.set i (slide b D STRIKE (p.getD i []))) pos) hit
    | _, _ => pos
  else pos

def strikeAll (b : Board) (np : Nat) (gates : List Gate) (pos : List (List Cell)) : List (List Cell) :=
  ((List.range gates.length).zip gates).foldl (fun p jg => strikeOne b np jg.1 jg.2 p) pos

theorem gate_keeps_heading (b : Board) (s s' : St) (a : Act) (h : step b s a = .ok s') :
    s'.gates.map (·.dir) = s.gates.map (·.dir) := by
  have hclick : ∀ x y, (clickSelect b s x y).gates = s.gates := by
    intro x y
    unfold clickSelect
    split
    · split <;> rfl
    · rfl
  have hzip : ∀ {α β : Type} (l : List α) (f : α → β),
      ((List.range l.length).zip l).map (fun p => f p.2) = l.map f := by
    intro α β l f
    have h2 : ((List.range l.length).zip l).map Prod.snd = l := List.map_snd_zip (by simp)
    calc ((List.range l.length).zip l).map (fun p => f p.2)
        = (((List.range l.length).zip l).map Prod.snd).map f := by rw [List.map_map]; rfl
      _ = l.map f := by rw [h2]
  have hmove : ∀ (g : Gate) (l : List Cell), (g.moveTo l).dir = g.dir := by
    intro g l; cases l <;> rfl
  have hadv : (advance b s a).gates.map (·.dir) = s.gates.map (·.dir) := by
    cases a with
    | click x y => simp only [advance, St.tock, hclick]
    | dir d =>
      simp only [advance, St.tock, tickGates, List.map_map]
      have : ∀ (jg : Nat × Gate) (pos : List (List Cell)) (n : Nat),
          ((Gate.tick (jg.2.moveTo (pos.getD (n + jg.1) []))).dir) = jg.2.dir := by
        intro jg pos n; exact hmove _ _
      simp only [Function.comp_def, this]
      exact hzip s.gates (·.dir)
  unfold step finish at h
  split at h
  · cases h
  · split at h
    · cases h
    · cases h
      exact hadv

What a pass establishes

consistent: for every recorded transition, as read by the agent's parser and with duplicates removed, every rule whose condition holds also has its effect hold. faithful: the certified run, recomputed step by step from the first screen of each level, matches the record and ends in a win. Axioms used: propext, all among Lean's standard ones.

What it does not establish: that the parser reads the screens correctly (that step runs in Python, outside Lean), or that a rule whose condition never occurs in the record is right. The Python file that produced the readings inside the model file is in the archive at meta/repair-20260928/lib/ka59-20260917/_lib.py.
