Push coloured objects around; the level is cleared when each ring centre is covered by a same-colour object.

8 levels · 12 rules · all levels replayed in Lean

How it plays

There are several coloured objects on the board (cross, diagonal cross, diamond, hollow square frame, horizontal line), and the white cursor sits at the centre of the one currently being controlled. The up, down, left and right keys push this whole object by 3 cells, and the switch key moves the cursor to another object. The dark small square frames are rings, and the colour at a ring's centre says which colour of object it is to be covered by. The solid-colour blocks in the grey frame are colour blocks, and an object that touches one turns entirely into this colour. The light-grey solid blocks are blockers and do not move: when a cross hits one, its two bars each go their own way; a hollow square frame that hits one changes shape, with the perimeter unchanged. The level is cleared when every ring centre is covered by an object of the same colour. The bottom row is the counter bar.

Replay

The certified run that cleared every level, one action at a time.

Keys: ← → step, Space play / pause. A click is circled; cells that changed are outlined.

Model

The final model: 7 object types and 12 rules. Each rule is a Python check against every recorded step; most are also Lean definitions.

Object types

  • Objects piecesPushable coloured shapes: cross, diagonal cross, diamond, hollow square frame, horizontal line.
  • Cursor cursorA white dot at the centre of the object being controlled.
  • Rings ringsDark small square frames; the centre colour says which object colour must cover it.
  • Colour blocks swatchesA solid colour in a grey frame; an object touching it turns this colour.
  • Blockers blockersLight-grey solid blocks; never move or get covered; objects hitting them deform.
  • Counter bar spentThe bottom row, recording how many cells this level has spent.
  • Stacking order overWhich of two overlapping objects is drawn on top.

def blockedCross (b : Board) (p : Piece) (a : Act) : Piece :=
  let dr := a.delta.1
  let dc := a.delta.2
  let hasH := decide (p.left < p.right)
  let hasV := decide (p.top < p.bottom)
  let hb := hasH && b.blockers.any (fun rc =>
    rc.1 - dr == p.hr && decide (p.left ≤ rc.2 - dc) && decide (rc.2 - dc ≤ p.right))
  let vb := hasV && b.blockers.any (fun rc =>
    rc.2 - dc == p.hc && decide (p.top ≤ rc.1 - dr) && decide (rc.1 - dr ≤ p.bottom))
  let rh := if hasH then (if hb then p.hr else p.hr + dr) else (if vb then p.hr else p.hr + dr)
  let l  := if hasH then (if hb then p.left else p.left + dc) else (if vb then p.hc else p.hc + dc)
  let rr := if hasH then (if hb then p.right else p.right + dc) else (if vb then p.hc else p.hc + dc)
  let cv := if hasV then (if vb then p.hc else p.hc + dc) else (if hb then p.hc else p.hc + dc)
  let t  := if hasV then (if vb then p.top else p.top + dr) else (if hb then p.hr else p.hr + dr)
  let bt := if hasV then (if vb then p.bottom else p.bottom + dr) else (if hb then p.hr else p.hr + dr)
  { p with top := min t rh, left := min l cv, bottom := max bt rh, right := max rr cv,
           hr := rh, hc := cv }

def blockedBox (p : Piece) (a : Act) : Piece :=
  let dr := a.delta.1
  let dc := a.delta.2
  let h := p.bottom - p.top + 1
  let w := p.right - p.left + 1
  if dc ≠ 0 then
    let l2 := if dc > 0 then p.left + 3 else p.left
    let r2 := if dc > 0 then p.right else p.right - 3
    let d1 := iabs ((p.top + (p.bottom + 3) + 1) / 2 - p.cr)
    let d2 := iabs (((p.top - 3) + p.bottom + 1) / 2 - p.cr)
    let t2 := if d2 < d1 then p.top - 3 else p.top
    let b2 := if d2 < d1 then p.bottom else p.bottom + 3
    if min (h + 3) (w - 3) < 4 then p
    else { p with top := t2, left := l2, bottom := b2, right := r2,
                  hr := (t2 + b2 + 1) / 2, hc := (l2 + r2 + 1) / 2 }
  else
    let t2 := if dr > 0 then p.top + 3 else p.top
    let b2 := if dr > 0 then p.bottom else p.bottom - 3
    let d1 := iabs ((p.left + (p.right + 3) + 1) / 2 - p.cc)
    let d2 := iabs (((p.left - 3) + p.right + 1) / 2 - p.cc)
    let l2 := if d2 < d1 then p.left - 3 else p.left
    let r2 := if d2 < d1 then p.right else p.right + 3
    if min (h - 3) (w + 3) < 4 then p
    else { p with top := t2, left := l2, bottom := b2, right := r2,
                  hr := (t2 + b2 + 1) / 2, hc := (l2 + r2 + 1) / 2 }

def Piece.touches (p : Piece) (s : Swatch) : Bool :=
  (List.range s.size.toNat).any (fun i =>
    (List.range s.size.toNat).any (fun j => p.mem (s.top + i) (s.left + j)))

def firstTouched (ss : List Swatch) (p : Piece) : Option Swatch :=
  (ss.filter (fun s => p.touches s)).foldl
    (fun acc s => match acc with
      | none => some s
      | some t => if s.top < t.top || (s.top == t.top && s.left < t.left) then some s else acc)
    none
