Move the player along bridges, avoiding sentries, patrols and chasers; walking onto the goal clears the level.

How it plays

The cells are connected by grey bridges. The blue one is the player, and the small mark on it shows its facing; the green cell is the goal. Press a direction key: if there is a bridge in that direction, the player moves over one cell, then each orange patrol walks one cell along its bridge (turning around at the end), and each awake dark-red chaser pounces one cell along the shortest path toward the cell the player was on; if there is no bridge in that direction, nobody moves. Walking onto the goal clears the level. Red sentries stand still and stare at the cell right in front of them: stepping into that cell from the front loses, while stepping into their cell from the side or from behind can trample them away; patrols and sleeping chasers can also be trampled from the side. Being landed on by a patrol or chaser loses, and if in the same step that the player walks onto the goal a patrol also lands on the goal, it likewise counts as a loss. A chaser wakes when the player walks to within two cells in front of it. The budget band along the bottom edge gets a notch shorter with each press, and reaching zero also counts as a loss.

Model

The final model: 8 object types and 10 rules. Each rule is a Python check against every recorded step; most are also Lean definitions.

Object types

  • Bridges bridgesWhich pairs of cells have a grey bridge between them and are passable.
  • Goal targetThe green cell; stepping onto it clears the level.
  • Player playerWhich cell the blue block stands on.
  • Facing facingWhich way the player's small mark points; after moving, it points the way just moved.
  • Sentries sentriesRed; stands still and stares at the cell right in front.
  • Patrols patrolsOrange; per real player step, walks one cell along its bridge, turning around at ends.
  • Chasers chasersDark red; still until awake, then pounces toward the player along the shortest path.
  • Budget band hudThe bottom row of the screen: how many steps remain this level.

def patrolNext (b : Board) (moved : Bool) (g : Guard) : Guard :=
  if moved then

def wakeDist : Nat := 2

def chaserNext (b : Board) (before after : Cell) (moved : Bool) (g : Guard) : Guard :=
  if moved then
    if g.alert then
      let c := if linked b.bridges g.cell g.facing then g.cell.move g.facing else g.cell
      { g with cell := c, facing := (firstStep b.bridges c before).getD g.facing }
    else if g.cell.dist after ≤ wakeDist then
      { g with alert := true, facing := (firstStep b.bridges g.cell before).getD g.facing }
    else g
  else g

theorem sentries_come_from (b : Board) (s s' : St) (d : Dir) (h : step b s d = .ok s') :
    ∀ g ∈ s'.sentries, g ∈ s.sentries ∧ stomped (landing b s d) d g = false := by
  rw [step_ok b s s' d h]
  intro g hg
  simp only [advance, sentriesNext, List.mem_filter, Bool.not_eq_true'] at hg
  exact hg

theorem patrols_come_from (b : Board) (s s' : St) (d : Dir) (h : step b s d = .ok s') :
    ∀ g ∈ s'.patrols, ∃ g0 ∈ s.patrols, stomped (landing b s d) d g0 = false ∧
      g = patrolNext b (landing b s d).isSome g0 := by

  intro g hg
  simp only [advance, patrolsNext, List.mem_map, List.mem_filter, Bool.not_eq_true'] at hg

def causes(w, d, level) -> list[str]:
    out = []
    h = hud_after(BUDGET[level], w.hud)
    q = landing(w, d)
    if q is not None:
        live_s = [g for g in w.sentries if not stomped(g, q, d)]
        live_p = [g for g in w.patrols if not stomped(g, q, d)]
        if any(g.cell + g.facing == q for g in live_s):
            out.append("front")
        if any(patrol_next(w, g, True).cell == q for g in live_p):
            out.append("patrol")
        if any(g.cell == q and g.cell + g.facing == w.player for g in live_p):
            out.append("swap")
        if any(chaser_next(w, g, w.player, q, True).cell == q for g in w.chasers):
            out.append("chaser")
    if h == 0:
        out.append("budget")
    return out

namespace Game

def dies (b : Board) (s : St) (d : Dir) : Bool :=
  let q := landing b s d
  let byGuard := match q with
    | none => false
    | some c =>
      (s.sentries.any fun g => !stomped q d g && g.cell.move g.facing == c)
      || (s.patrols.any fun g => !stomped q d g && (patrolNext b true g).cell == c)
      || (s.patrols.any fun g => !stomped q d g && g.cell == c && g.cell.move g.facing == s.player)
      || (s.chasers.any fun g => (chaserNext b s.player c true g).cell == c)
  byGuard || runsOut b s

end Game
