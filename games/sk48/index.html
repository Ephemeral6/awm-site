Stretch and slide ropes so each rope's target blocks match the colour order shown in the diagram.

8 levels · 7 rules · all levels replayed in Lean

How it plays

There are two kinds of player: the horizontal player hangs on the vertical track on the left and extends a horizontal rope to the right; the vertical player hangs on the horizontal track at the top and extends a vertical rope downward. Click a player to select it. Pressing a key along the rope's direction lengthens or shortens the rope by one cell, the coloured target blocks strung on the rope move along, and the run in front of the rope tip is pushed away; pressing a key perpendicular to the rope translates the player together with the whole rope by one cell, and the target blocks the rope sweeps over are carried away. When the colour order of each rope, counted outward from the player, agrees with the corresponding group in the diagram at the bottom of the screen, the level is cleared. Things cannot be pushed when they hit the board edge, a wall, or are pinned by another rope; dead ropes hanging from the top are also named in the diagram.

Replay

The certified run that cleared every level, one action at a time. Beside each step: the rules of the model that explain it.

Model

The final model: 6 object types and 7 rules. Each rule is a Python check against every recorded step; most are also Lean definitions.

Object types

  • Board boardFixed per level: grid size, walls, two tracks, the diagram, and dead ropes.
  • Target blocks blocksEach coloured block's cell and colour on the board.
  • Horizontal player hplayerHangs on the left track with a horizontal rope; records its row and rope length.
  • Vertical players vplayersHang on the top track with vertical ropes; record each column and rope length.
  • Selection selectedWhich player the direction keys currently act on.
  • Bottom bar barThe bottom line of the screen, shortening right to left as direction keys are pressed.

def predict(w, a):
    from derive.line import key_axis, mover, pins
    from objects import Block, Cell, HPlayer, VPlayer
    mv = mover(w)
    _, k = key_axis(mv, a.kind)
    walls = w.wall_set()
    m = w.block_map()
    pin = pins(w, mv)
    L, n, at = mv.L, mv.n, mv.at
    if k > 0 and (n >= L or at(n) in walls):
        return w
    if k < 0 and n == 0:
        return w
    occ = [i for i in range(min(n, L)) if at(i) in m]
    run = []
    if k > 0:
        i = n
        while i < L and at(i) in m:
            run.append(i)
            i += 1
    run_ok = bool(run) and not any(at(i) in pin for i in run) and \
        run[-1] + 1 < L and at(run[-1] + 1) not in pin and at(run[-1] + 1) not in walls
    moving_run = run if run_ok else []
    moving: list[int] = []
    for i in (reversed(occ) if k > 0 else occ):
        t = i + k
        if at(i) in pin or not (0 <= t < L) or at(t) in pin:
            continue
        if at(t) in m and t not in moving and t not in moving_run:
            continue
        moving.append(i)
    go = set(moving_run) | set(moving)
    nm = {}
    for (r, c), col in m.items():
        i = c if mv.orient == "h" else r
        on_line = (r if mv.orient == "h" else c) == mv.pos
        nm[at(i + k) if (on_line and i in go) else (r, c)] = col
    blocks = frozenset(Block(Cell(r, c), col) for (r, c), col in nm.items())
    if mv.orient == "h":
        return dataclasses.replace(w, blocks=blocks, hplayer=HPlayer(mv.owner, mv.pos, n + k))
    vs = tuple(VPlayer(v.owner, v.col, n + k) if v.owner == mv.owner else v for v in w.vplayers)
    return dataclasses.replace(w, blocks=blocks, vplayers=vs)

def assert_(prev, action, nxt, ctx):

    w, nw = parse(prev), parse(nxt)

def runFrom (s : St) (ln : Line) : Nat → Nat → List Nat
  | _, 0 => []
  | i, fuel + 1 => if i < ln.len && s.has (ln.cellAt i) then i :: runFrom s ln (i + 1) fuel else []

def blocked (s : St) (ln : Line) (pins : List Cell) (acc run : List Nat) (t : Nat) : Bool :=
  !(t < ln.len) || pins.contains (ln.cellAt t) ||
    (s.has (ln.cellAt t) && !(acc.contains t) && !(run.contains t))

def ropeCore (walls pins : List Cell) (s : St) (ln : Line) (n : Nat) (ext : Bool) : Option (List Block) :=
  if ext && (decide (ln.len ≤ n) || walls.contains (ln.cellAt n)) then none
  else if !ext && n == 0 then none
  else
    let occ := (List.range (min n ln.len)).filter (fun i => s.has (ln.cellAt i))
    let run := if ext then runFrom s ln n ln.len else []
    let runOk : Bool :=
      match run.getLast? with
      | none => false
      | some last =>
        !(run.any (fun i => pins.contains (ln.cellAt i))) && last + 1 < ln.len &&
          !(pins.contains (ln.cellAt (last + 1))) && !(walls.contains (ln.cellAt (last + 1)))
    let movingRun := if runOk then run else []
    let order := if ext then occ.reverse else occ
    let moving := order.foldl (fun acc i =>
        if pins.contains (ln.cellAt i) then acc
        else if ext then (if blocked s ln pins acc movingRun (i + 1) then acc else acc ++ [i])
        else if i == 0 then acc
        else if blocked s ln pins acc movingRun (i - 1) then acc else acc ++ [i]) []
    let go := movingRun ++ moving
    some (s.blocks.map (fun bl =>
      match go.find? (fun i => ln.cellAt i == bl.cell) with
      | some i => { bl with cell := ln.cellAt (if ext then i + 1 else i - 1) }
      | none => bl))

def stepLen (ext : Bool) (n : Nat) : Nat := if ext then n + 1 else n - 1

def ropeSlideV (b : Board) (s : St) (ext : Bool) : St :=
  match s.vSel with
  | some v =>
    match ropeCore b.walls (pinsOf b s v.owner) s ⟨fun i => ⟨i, v.col⟩, b.grid.rows⟩ v.rope ext with
    | some bl => { s with blocks := bl,
                          vs := s.vs.map (fun x => if x.owner == v.owner then { x with rope := stepLen ext x.rope } else x) }
    | none => s
  | none => s

def ropeSlide (b : Board) (s : St) (ext : Bool) : St :=
  match s.h with
  | some h =>
    if h.owner == s.sel then
      match ropeCore b.walls (pinsOf b s h.owner) s ⟨fun i => ⟨h.row, i⟩, b.grid.cols⟩ h.rope ext with
      | some bl => { s with blocks := bl, h := some { h with rope := stepLen ext h.rope } }
      | none => s
    else ropeSlideV b s ext
  | none => ropeSlideV b s ext

def predict(w, a):
    from derive.line import key_axis, mover
    from objects import Block, Cell, HPlayer, VPlayer
    mv = mover(w)
    _, d = key_axis(mv, a.kind)
    walls = w.wall_set()
    m = w.block_map()
    t = mv.pos + d
    if not (0 <= t < mv.W) or not (mv.rail_lo <= t <= mv.rail_hi):
        return w
    pole = range(min(mv.n, mv.L))
    if any(mv.at(i, t) in walls for i in pole):
        return w

    def shift(cell):
        r, c = cell
        return (r + d, c) if mv.orient == "h" else (r, c + d)

    cells: list[tuple[int, int]] = []
    for i in pole:
        here, there = mv.at(i), mv.at(i, t)
        s = here if here in m else (there if there in m else None)
        while s is not None and s in m:
            cells.append(s)
            s = shift(s)
    g = w.board.grid
    if any(not g.inside(*shift(x)) for x in cells):
        return w
    go = set(cells)
    blocks = frozenset(Block(Cell(*(shift(x) if x in go else x)), col) for x, col in m.items())
    if mv.orient == "h":
        return dataclasses.replace(w, blocks=blocks, hplayer=HPlayer(mv.owner, t, mv.n))
    vs = tuple(sorted((VPlayer(v.owner, t, v.rope) if v.owner == mv.owner else v for v in w.vplayers),
                      key=lambda v: v.col))
    return dataclasses.replace(w, blocks=blocks, vplayers=vs)

def chainFrom (s : St) (dest : Cell → Option Cell) : Cell → Nat → List Cell
  | _, 0 => []
  | x, fuel + 1 =>
    if s.has x then
      x :: (match dest x with
            | some y => chainFrom s dest y fuel
            | none => [])
    else []

def sweepCore (b : Board) (s : St) (pole : List Cell) (dest : Cell → Option Cell) : Option (List Block) :=
  let fuel := b.grid.rows + b.grid.cols
  let cells := pole.flatMap (fun p =>
    if s.has p then chainFrom s dest p fuel
    else match dest p with
      | some q => if s.has q then chainFrom s dest q fuel else []
      | none => [])
  if cells.any (fun x => (dest x).isNone) then none
  else some (s.blocks.map (fun bl =>
    if cells.contains bl.cell then
      match dest bl.cell with
      | some y => { bl with cell := y }
      | none => bl
    else bl))

def downOrUp (rows : Nat) (fwd : Bool) (c : Cell) : Option Cell :=
  if fwd then (if c.row + 1 < rows then some ⟨c.row + 1, c.col⟩ else none)
  else (if c.row == 0 then none else some ⟨c.row - 1, c.col⟩)

def rightOrLeft (cols : Nat) (fwd : Bool) (c : Cell) : Option Cell :=
  if fwd then (if c.col + 1 < cols then some ⟨c.row, c.col + 1⟩ else none)
  else (if c.col == 0 then none else some ⟨c.row, c.col - 1⟩)

def railOk (r : Option Rail) (k : Nat) : Bool :=
  match r with
  | some r => r.covers k
  | none => false

def poleSweepV (b : Board) (s : St) (fwd : Bool) : St :=
  match s.vSel with
  | some v =>
    match rightOrLeft b.grid.cols fwd ⟨0, v.col⟩ with
    | none => s
    | some t0 =>
      if !(railOk b.vrail t0.col) || (vCells v.col v.rope b.grid.rows).any (fun c => b.isWall ⟨c.row, t0.col⟩) then s
      else match sweepCore b s (vCells v.col v.rope b.grid.rows) (rightOrLeft b.grid.cols fwd) with
        | some bl => { s with blocks := bl,
                              vs := s.vs.map (fun x => if x.owner == v.owner then { x with col := t0.col } else x) }
        | none => s
  | none => s

def poleSweep (b : Board) (s : St) (fwd : Bool) : St :=
  match s.h with
  | some h =>
    if h.owner == s.sel then
      match downOrUp b.grid.rows fwd ⟨h.row, 0⟩ with
      | none => s
      | some t0 =>
        if !(railOk b.hrail t0.row) || (hCells h.row h.rope b.grid.cols).any (fun c => b.isWall ⟨t0.row, c.col⟩) then s
        else match sweepCore b s (hCells h.row h.rope b.grid.cols) (downOrUp b.grid.rows fwd) with
          | some bl => { s with blocks := bl, h := some { h with row := t0.row } }
          | none => s
    else poleSweepV b s fwd
  | none => poleSweepV b s fwd
