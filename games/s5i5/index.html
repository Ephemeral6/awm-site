How it plays

There are several coloured long bars on the board, with a grey cap at one end and the other end able to extend and retract; the cap of one bar hangs on another bar, strung into an arm, and the bar at the end carries a dark-red tip. The panels in the lower row each govern one colour: click the left icon and every bar of this colour retracts by 3 cells, click the right one and it extends by 3 cells; the bar that would press on the panel stays in place, and the others move as usual; the bars or walls hanging on the extending/retracting end are pushed away or dragged back along. Clicking a legend frame in the upper row makes the bars of this colour, together with the whole arm hanging behind it, rotate counterclockwise about the joint. The level is cleared when all the tips land on the centres of the diamond targets. In many cases this step is voided entirely: two things fight for the same cell, pressing on a purple fixed wall, retracting until the body is not enough, a pushed-away wall has nowhere to land, being unable to rotate through (pressing on the panel while rotating also counts).

Model

The final model: 5 object types and 19 rules. Each rule is a Python check against every recorded step; 17 of them are translated into Lean (the paper's per-game table). Code under a rule is the file of the same name in the agent's Lean project.

def dragBars (col : Nat) (d : Dir) (f : Rect) :
    List (Bar × Mark) → List (Bar × Mark) × List (Dir × Rect × Nat)
  | [] => ([], [])
  | (b, m) :: rest =>
    let r := dragBars col d f rest
    if m.isStill && b.color != col && b.cap.meets f then
      ((b, .moved d) :: r.1, (d, b.face, b.color) :: r.2)
    else ((b, m) :: r.1, r.2)

def dragWalls (d : Dir) (f : Rect) : List (Wall × Option Dir) → List (Wall × Option Dir)
  | [] => []
  | (w, m) :: rest =>
    (if m.isNone && w.rect.meets f then (w, some d) else (w, m)) :: dragWalls d f rest

def follow : Nat → List (Dir × Rect × Nat) → List (Bar × Mark) → List (Wall × Option Dir) →
    List (Bar × Mark) × List (Wall × Option Dir)
  | 0, _, bs, ws => (bs, ws)
  | _ + 1, [], bs, ws => (bs, ws)
  | n + 1, (d, f, col) :: st, bs, ws =>
    let r := dragBars col d f bs
    follow n (r.2.reverse ++ st) r.1 (dragWalls d f ws)

def followFrom (fuel : Nat) : List (Bar × Mark) → List (Bar × Mark) → List (Wall × Option Dir) →
    List (Bar × Mark) × List (Wall × Option Dir)
  | [], bs, ws => (bs, ws)
  | (b, .grown d) :: rest, bs, ws =>
    let r := follow fuel [(d, b.face, b.color)] bs ws

  | _ :: rest, bs, ws => followFrom fuel rest bs ws

def turnBar1 (pr pc : Nat) (b : Bar) : Bar :=
  let cap := b.cap.rot pr pc
  { b with r := cap.r0, c := cap.c0, dir := b.dir.ccw }

def turnBar (pr pc : Nat) : Nat → Bar → Bar
  | 0, b => b
  | k + 1, b => turnBar pr pc k (turnBar1 pr pc b)

def turnAll (pr pc k : Nat) : List (Bar × Bool) → List Bar
  | [] => []
  | (b, m) :: rest => (if m then turnBar pr pc k b else b) :: turnAll pr pc k rest

def jointOf (b : Bar) : Nat × Nat :=
  match b.dir with
  | .U => (b.r - 1, b.c + 1)
  | .D => (b.r + 1, b.c + 1)
  | .L => (b.r + 1, b.c - 1)
  | .R => (b.r + 1, b.c + 1)

def pivotRoots (w : World) (K : Nat) : Option (Nat × Nat × List Bool) :=
  match w.board.hubs.find? (·.color == K) with
  | some h => some (h.r + 1, h.c + 1, w.bars.map (fun b => meetsAny b.cap h.ring))
  | none =>
    match w.bars.filter (·.color == K) with
    | [b] => some ((jointOf b).1, (jointOf b).2, w.bars.map (·.color == K))
    | _ => none

def turnInfo (w : World) (pr pc k : Nat) (chain : List (Bar × Bool)) : Bool × Bool × Bool × Bool :=
  let moved := (chain.filter (·.2)).map (fun p => turnBar pr pc k p.1)
  let others := (chain.filter (fun p => !p.2)).map (fun p => p.1.whole)
              ++ w.board.hubs.map Hub.rect ++ w.walls.map (·.rect) ++ w.board.boxes
  (moved.any (fun b => meetsAny b.whole w.board.fixed),
   moved.any (fun b => meetsAny b.whole others),
   moved.any (fun b => !b.whole.onBoard),
   moved.any (fun b => !b.cap.onBoard))

def tryTurn (w : World) (pr pc : Nat) (chain : List (Bar × Bool)) : Nat → Nat → Res
  | _, 0 => .refused
  | k, n + 1 =>
    match turnInfo w pr pc k chain with
    | (fx, bl, off, co) =>
      match verdict k fx bl off co with
      | .refuse => .refused
      | .next => tryTurn w pr pc chain (k + 1) n
      | .accept => .ok { w with bars := turnAll pr pc k chain }

def rotate (w : World) (K : Nat) : Res :=
  match pivotRoots w K with
  | none => .silent
  | some (pr, pc, roots) => tryTurn w pr pc (w.bars.zip (downstream w.bars roots)) 1 3

What a pass establishes

consistent: for every recorded transition, as read by the agent's parser and with duplicates removed, every rule whose condition holds also has its effect hold. faithful: the certified run, recomputed step by step from the first screen of each level, matches the record and ends in a win. Axioms used: propext, all among Lean's standard ones.

What it does not establish: that the parser reads the screens correctly (that step runs in Python, outside Lean), or that a rule whose condition never occurs in the record is right.
