How it plays

On the screen there are a line of question text, a legend book and an answer bar. Each rule in the legend is a pair of glyph strings: the key string on the left is replaced by the value string on the right. Rewrite the question text segment by segment according to the legend, sometimes following the colour chain through several rounds, until it is rewritten into the colour of the answer bar and exactly fills the answer bar; what is obtained is the answer. The left and right keys move the focus (the string boxed by the corner markers) to the next string or the previous string; the up and down keys turn the shape of every glyph in the focus string forward or backward by one step along a ring of seven. When the answer bar equals the answer shape cell by cell, the level is cleared; only the shape is looked at, not the orientation. In some levels the focus rests on the legend, and what must be rewritten is the legend itself. The counter bar at the very bottom eats up one cell every few steps.

Model

The final model: 6 object types and 7 rules. Each rule is a Python check against every recorded step; 4 of them are translated into Lean (the paper's per-game table). Code under a rule is the file of the same name in the agent's Lean project.

def shiftShape (s : Shape) (up : Bool) : Shape :=
  s / 7 * 7 + (if up then (s % 7 + 1) % 7 else (s % 7 + 6) % 7)

def modifyAt {α : Type} : List α → Nat → (α → α) → List α
  | [], _, _ => []
  | x :: xs, 0, f => f x :: xs
  | x :: xs, n + 1, f => x :: modifyAt xs n f

def turnRun (r : Run) (up : Bool) : Run := { r with shapes := r.shapes.map (fun s => shiftShape s up) }

def turnRule (i : Focus) (up : Bool) (r : LegendRule) : LegendRule :=
  if i % 2 == 0 then { r with key := turnRun r.key up } else { r with val := turnRun r.val up }

def phaseStep (w : World) (up : Bool) : St :=
  if w.board.legendMode then
    { w.st with legend := modifyAt w.st.legend (w.st.focus / 2) (turnRule w.st.focus up) }
  else
    { w.st with answer := modifyAt w.st.answer w.st.focus (fun s => shiftShape s up) }

def rewriteAll (rules : List (List Shape × List Shape)) : Nat → List Shape → List (List Shape)
  | _, [] => [[]]
  | 0, _ :: _ => []
  | fuel + 1, s :: ss =>
      rules.flatMap (fun kv =>
        if !kv.1.isEmpty && kv.1.isPrefixOf (s :: ss) then
          (rewriteAll rules fuel ((s :: ss).drop kv.1.length)).map (fun rest => kv.2 ++ rest)
        else [])

def colorPairs (l : Legend) : List (Nat × Nat) := (l.map (fun r => (r.key.color, r.val.color))).eraseDups

def groupRules (l : Legend) (p : Nat × Nat) : List (List Shape × List Shape) :=
  (l.filter (fun r => r.key.color == p.1 && r.val.color == p.2)).map (fun r => (r.key.shapes, r.val.shapes))

def chainRound (l : Legend) (ac : Nat) (fr : List (Nat × List Shape)) :
    List (List Shape) × List (Nat × List Shape) :=
  (fr.filterMap (fun cs => if cs.1 == ac then some cs.2 else none),
   fr.flatMap (fun cs =>
     if cs.1 == ac then []
     else (colorPairs l).flatMap (fun p =>
       if p.1 == cs.1 then (rewriteAll (groupRules l p) cs.2.length cs.2).map (fun r => (p.2, r))
       else [])))

def chain (l : Legend) (ac : Nat) : Nat → List (Nat × List Shape) → List (List Shape)
  | 0, _ => []
  | k + 1, fr =>
      match chainRound l ac fr with
      | (out, nxt) => out ++ chain l ac k nxt

def legendColors (l : Legend) : List Nat := ((colorPairs l).flatMap (fun p => [p.1, p.2])).eraseDups

def targets (w : World) : List (List Shape) :=
  chain w.st.legend w.board.ansColor ((legendColors w.st.legend).length + 1)
    [(w.board.prompt.color, w.board.prompt.shapes)]

def isGoal (w : World) : Bool := (targets w).contains w.st.answer

def _others(w, skip):
    return tuple(s.glyphs for i, s in enumerate(w.stops()) if i != skip)

def _layout(w):
    runs = [w.prompt, w.answer] + [r for x in w.legend.rules for r in (x.key, x.val)]
    return tuple(r.color for r in runs)

def assert_(prev, action, nxt, ctx=None):
    if scope_(prev, action, nxt, ctx):
        return "NA"
    w, v = world(prev), world(nxt)
    if w is None or v is None:
        return "NA"
    if v.prompt != w.prompt or v.backdrop != w.backdrop or _layout(v) != _layout(w):
        return "FALSE"
    if v.focus.kind != w.focus.kind:
        return "FALSE"
    if action in MOVE:
        same = (v.answer == w.answer and v.legend == w.legend)
        return "TRUE" if same else "FALSE"
    if action in TURN:
        if v.focus != w.focus:
            return "FALSE"
        i = w.focus.index
        other_same = (v.legend == w.legend) if w.focus.kind == "answer" else (v.answer == w.answer)
        return "TRUE" if other_same and _others(v, i) == _others(w, i) else "FALSE"
    return "FALSE"

theorem step_ok {w : World} {a : Act} {s : St} (h : step w a = .ok s) : s = after w a := by
  unfold step at h
  split at h
  · cases h
  · cases h; rfl

theorem frameRest_move (w : World) (a : Act) (s : St) (ha : a = .a3 ∨ a = .a4)
    (h : step w a = .ok s) : s.answer = w.st.answer ∧ s.legend = w.st.legend := by
  rw [step_ok h]
  rcases ha with rfl | rfl <;> exact ⟨rfl, rfl⟩

theorem frameRest_turn_focus (w : World) (a : Act) (s : St) (ha : a = .a1 ∨ a = .a2)
    (h : step w a = .ok s) : s.focus = w.st.focus := by
  rw [step_ok h]
  rcases ha with rfl | rfl <;> simp only [after, phaseStep] <;> split <;> rfl

theorem frameRest_turn_answer (w : World) (a : Act) (s : St) (ha : a = .a1 ∨ a = .a2)
    (hm : w.board.legendMode = false) (h : step w a = .ok s) :
    s.legend = w.st.legend ∧ ∀ i, i ≠ w.st.focus → s.answer[i]? = w.st.answer[i]? := by
  have ne : ∀ (l : List Shape) (n i : Nat) (f : Shape → Shape), i ≠ n → (modifyAt l n f)[i]? = l[i]? := by
    intro l
    induction l with
    | nil => intro n i f _; rfl
    | cons x xs ih =>
      intro n i f hi
      cases n with
      | zero => cases i with
        | zero => exact absurd rfl hi
        | succ i => rfl
      | succ n => cases i with
        | zero => rfl
        | succ i =>
          simp only [modifyAt, List.getElem?_cons_succ]
          exact ih n i f (fun e => hi (by rw [e]))
  rw [step_ok h]
  constructor
  · rcases ha with rfl | rfl <;> simp [after, phaseStep, hm]
  · intro i hi
    rcases ha with rfl | rfl <;>
      simp only [after, phaseStep, hm, Bool.false_eq_true, ↓reduceIte] <;>
      exact ne _ _ _ _ hi

theorem frameRest_turn_legend (w : World) (a : Act) (s : St) (ha : a = .a1 ∨ a = .a2)
    (hm : w.board.legendMode = true) (h : step w a = .ok s) :
    s.answer = w.st.answer ∧ ∀ k, k ≠ w.st.focus / 2 → s.legend[k]? = w.st.legend[k]? := by
  have ne : ∀ (l : List LegendRule) (n i : Nat) (f : LegendRule → LegendRule), i ≠ n →
      (modifyAt l n f)[i]? = l[i]? := by

theorem frameRest_colors (w : World) (a : Act) (s : St) (h : step w a = .ok s) :
    s.legend.map (fun r => (r.key.color, r.val.color)) =
      w.st.legend.map (fun r => (r.key.color, r.val.color)) := by
  have keep : ∀ (l : List LegendRule) (n : Nat) (f : LegendRule → LegendRule),
      (∀ r, (f r).key.color = r.key.color ∧ (f r).val.color = r.val.color) →
      (modifyAt l n f).map (fun r => (r.key.color, r.val.color)) = l.map (fun r => (r.key.color, r.val.color)) := by
    intro l
    induction l with
    | nil => intro n f _; rfl
    | cons x xs ih =>
      intro n f hf
      cases n with
      | zero => simp [modifyAt, (hf x).1, (hf x).2]
      | succ n =>
        simp only [modifyAt, List.map_cons]
        rw [ih n f hf]
  have tr : ∀ (i : Nat) (up : Bool) (r : LegendRule),
      (turnRule i up r).key.color = r.key.color ∧ (turnRule i up r).val.color = r.val.color := by
    intro i up r
    unfold turnRule
    split <;> exact ⟨rfl, rfl⟩
  rw [step_ok h]
  cases a <;> simp only [after] <;> try rfl
  all_goals
    simp only [phaseStep]
    split
    · exact keep _ _ _ (tr _ _)
    · rfl

What a pass establishes

consistent: for every recorded transition, as read by the agent's parser and with duplicates removed, every rule whose condition holds also has its effect hold. faithful: the certified run, recomputed step by step from the first screen of each level, matches the record and ends in a win. Axioms used: propext, all among Lean's standard ones.

What it does not establish: that the parser reads the screens correctly (that step runs in Python, outside Lean), or that a rule whose condition never occurs in the record is right. The Python file that produced the readings inside the model file is in the archive at meta/repair-20260928/lib/tr87-20260907/_lib.py.
