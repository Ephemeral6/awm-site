How it plays

There are several groups of diamonds on the board, each group leading a jar, and the jar always stays at the centre of this group of diamonds. The jar cannot be clicked, only the diamonds can be moved: first click a diamond to select it, then click a piece of empty ground, it jumps over, and the jar immediately moves to the new centre. When a jar lands on a paint block, it covers itself with the colour of that block, and the paint block disappears with it. When the colour on the jar matches the colour a target ring wants and it presses on that ring's target centre, the target ring locks. The level is cleared when each group's jar presses on a locked target ring. Grey hard walls cannot be stepped on by anyone, and light-blue soft-ground diamonds can be stood on but a jar cannot press onto them; a move that would step on these places is voided. The vertical bar on the left records how many clicks have been made.

Model

The final model: 6 object types and 9 rules. Each rule is a Python check against every recorded step; 8 of them are translated into Lean (the paper's per-game table). Code under a rule is the file of the same name in the agent's Lean project.

def assert_(w, action, nw, ctx):
    got = {tuple(sorted(tuple(a) for a in g.anchors)): g.ink for g in nw.groups}
    moved = _landing(w, action, nw, ctx)
    k = blob_at(w, moved[2]) if moved else None
    for i, g in enumerate(w.groups):
        key = (tuple(tuple(x) for x in moved[1]) if moved and moved[0] == i
               else tuple(sorted(tuple(a) for a in g.anchors)))
        want = overwrite(g.ink, w.blobs[k].ink) if (k is not None and moved[0] == i) else g.ink
        if key not in got or not ink_agrees(want, got[key]):
            return False
    if k is None:
        return same_blobs(w, nw)
    rest = [b for j, b in enumerate(w.blobs) if j != k]
    return ([tuple(b.center) for b in rest] == [tuple(b.center) for b in nw.blobs]
            and all(ink_agrees(a.ink, b.ink) for a, b in zip(rest, nw.blobs)))

def overwrite (ink blob : List Nat) : List Nat :=
  List.zipWith (fun a b => if b = 0 then a else b) ink blob

def absorb (w : World) (i : Nat) (c : Cell) : World :=
  match w.blobs.find? (fun x => x.center = c) with
  | none => w
  | some x =>
    { w with
      groups := mapAt (fun g => { g with ink := overwrite g.ink x.ink }) i w.groups
      blobs := w.blobs.filter (fun y => y.center ≠ c) }

theorem terrain_static (b₁ b₂ : Board)
    (h : ∀ q, b₁.wallAt q = b₂.wallAt q ∧ b₁.softAt q = b₂.softAt q) (w : World) (p : Cell) :
    step b₁ w p = step b₂ w p := by
  have hb : ∀ c, blocked b₁ p c = blocked b₂ p c := by
    intro c
    have hw : (fun q => b₁.wallAt q) = (fun q => b₂.wallAt q) := funext fun q => (h q).1
    have hs : (fun q => b₁.wallAt q || b₁.softAt q) = (fun q => b₂.wallAt q || b₂.softAt q) :=
      funext fun q => by rw [(h q).1, (h q).2]
    unfold blocked
    rw [hw, hs]
  unfold step
  simp only [hb]

def expected(w, action, nw, ctx):
    rets = [(tuple(t.center), t.locked) for t in w.reticles]
    if _scope.on_diamond(w, action, ctx) is None:
        return sorted(rets)
    _, b = _scope.legality(w, action, nw, ctx)
    if b:
        return sorted(rets)
    i, _, c = placed(w, click_cell(action))
    ink = w.groups[i].ink
    k = blob_at(w, c)
    if k is not None:
        ink = overwrite(ink, w.blobs[k].ink)
    j = reticle_for(w, c, ink)
    if j is not None:
        rets[j] = (rets[j][0], True)
    return sorted(rets)

def scope_(w, action, nw, ctx):
    why = _scope.in_level(action, ctx)
    if why:
        return why
    if _scope.on_diamond(w, action, ctx) is None:
        return None
    why, _ = _scope.legality(w, action, nw, ctx)
    return why

def assert_(w, action, nw, ctx):
    return expected(w, action, nw, ctx) == sorted((tuple(t.center), t.locked) for t in nw.reticles)

def all_home(w, action, nw, ctx) -> bool:
    locked = {tuple(t.center) for t in w.reticles if t.locked}
    centers = [tuple(center(g)) for g in w.groups]
    why, b = _scope.legality_any(w, action, nw, ctx)
    if why is None and not b:
        i, _, c = placed(w, click_cell(action))
        ink = w.groups[i].ink
        k = blob_at(w, c)
        if k is not None:
            ink = overwrite(ink, w.blobs[k].ink)
        if reticle_for(w, c, ink) is not None:
            locked.add(tuple(c))
        centers[i] = tuple(c)
    return all(c in locked for c in centers)

Re-run the proof

Before you start: our run of this check took 252 s and 16.8 GB of memory at its peak (2026-10-24).

  1. Download r11l.zip and unpack it. The model and its theorems are in archive/r11l-20260907/formal/Game.lean. Before changing it, check that its SHA-256 is
    9ce832789da3054c5316e1f8848a8eb309d3ad5137cc7006eed99f264c721ed4
  2. Install Lean with elan, then the version named in the archive's lean-toolchain:
    elan toolchain install leanprover/lean4:v4.34.0
  3. Copy the model file to Check.lean and append these two lines to the copy:
    #print axioms consistent
#print axioms faithful
  4. In the folder that holds Check.lean, run:
    lean +leanprover/lean4:v4.34.0 -M60000 Check.lean
  5. Lean should finish without errors and print:
    'consistent' depends on axioms: [propext]
'faithful' depends on axioms: [propext]

What a pass establishes

consistent: for every recorded transition, as read by the agent's parser and with duplicates removed, every rule whose condition holds also has its effect hold. faithful: the certified run, recomputed step by step from the first screen of each level, matches the record and ends in a win. Axioms used: propext, all among Lean's standard ones.

What it does not establish: that the parser reads the screens correctly (that step runs in Python, outside Lean), or that a rule whose condition never occurs in the record is right. The Python file that produced the readings inside the model file is in the archive at meta/repair-20260928/lib/r11l-20260907/_lib.py.
