"""Emit lean/FinalTables.lean: the certificate tables as a balanced binary search tree."""
import pickle, sys
sys.path.insert(0, 'work/final')
D, T = pickle.load(open('work/final/tables.pkl', 'rb'))
def idx(q):
    s, r, c, op, it = q
    return ((4 if s == 4 else 0) + (2 if op else 0) + (1 if it else 0)) * 4096 + r * 64 + c
items = sorted((idx(q), tuple(T[k][q] for k in ('AN0','BS','BW','AM0','BMS','BMW'))) for q in D)
def build(lo, hi):
    if lo >= hi: return 'BT.leaf'
    mid = (lo + hi) // 2
    k, v = items[mid]
    return '(BT.node %d (%s) %s %s)' % (k, ', '.join(map(str, v)), build(lo, mid), build(mid + 1, hi))
sys.setrecursionlimit(100000)
body = build(0, len(items))
out = '''/-! Certificate tables for lean/FinalProof.lean, generated by work/final/emit.py from the
value iteration in work/final/gen.py. Their correctness is NOT assumed: FinalProof checks every
local inequality they must satisfy with `decide +kernel`. -/
namespace Final

/-- binary search tree keyed by `idx q`; value = (AN0, BS, BW, AM0, BMS, BMW). -/
inductive BT where
  | leaf
  | node (key : Nat) (v : Nat × Nat × Nat × Nat × Nat × Nat) (l r : BT)

def INF : Nat := 200

def BT.find : BT → Nat → Nat × Nat × Nat × Nat × Nat × Nat
  | .leaf, _ => (INF, INF, INF, INF, INF, INF)
  | .node k v l r, x => if x < k then l.find x else if k < x then r.find x else v

set_option maxRecDepth 200000 in
def tables : BT := %s

end Final
''' % body
open('lean/FinalTables.lean', 'w', encoding='utf-8').write(out)
print('entries', len(items), 'bytes', len(out))
