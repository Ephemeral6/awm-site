"""Emit lean/FinalCert.lean: the certificate check split into 320 small kernel-checked chunks
(one per size/opened/item/row), plus dispatch lemmas, so that peak memory stays small."""
L = ['import FinalModel', '', '/-! Certificate check, split into per-row chunks (generated by work/final/emit_cert.py). -/',
     '', 'namespace Final', '',
     '/-- all positions of one row (fixed size, opened, item) satisfy `facts` if in the domain. -/',
     'def chunk (s : Nat) (op it : Bool) (r : Nat) : Bool :=',
     '  (List.range 64).all fun c => !Dom { s := s, r := r, c := c, op := op, it := it } ||',
     '    facts { s := s, r := r, c := c, op := op, it := it }', '']
def nm(s, op, it, r): return 'ch_%d_%s_%s_%d' % (s, 't' if op else 'f', 't' if it else 'f', r)
for s in (2, 4):
    for op in (False, True):
        for it in (False, True):
            for r in range(40):
                L.append('theorem %s : chunk %d %s %s %d = true := by decide +kernel' %
                         (nm(s, op, it, r), s, str(op).lower(), str(it).lower(), r))
L.append('')
for s in (2, 4):
    for op in (False, True):
        for it in (False, True):
            L.append('theorem rows_%d_%s_%s (r : Nat) (hr : r < 40) : chunk %d %s %s r = true := by' %
                     (s, 't' if op else 'f', 't' if it else 'f', s, str(op).lower(), str(it).lower()))
            L.append('  match r, hr with')
            for r in range(40):
                L.append('  | %d, _ => exact %s' % (r, nm(s, op, it, r)))
            L.append('  | n + 40, h => exact absurd h (by omega)')
            L.append('')
L.append('theorem chunk_ok (s : Nat) (op it : Bool) (r : Nat) (hs : s = 2 ∨ s = 4) (hr : r < 40) :')
L.append('    chunk s op it r = true := by')
L.append('  rcases hs with rfl | rfl <;> cases op <;> cases it')
cases = []
for s in (2, 4):
    for op in (False, True):
        for it in (False, True):
            cases.append('  · exact rows_%d_%s_%s r hr' % (s, 't' if op else 'f', 't' if it else 'f'))
L += cases
L += ['', 'end Final', '']
open('lean/FinalCert.lean', 'w', encoding='utf-8').write('\n'.join(L))
print('written', len(L), 'lines')
