* State: block size (big 4x4 / small 2x2), top-left (r, c) on the odd lattice, barrier opened?, pad used?.
* Map: rectangles read off the level-4 frame (room, stalk, corridor, exit box; barrier cells and destroyed
  switch-box cells are floor once opened [a3 tr127]).
* Moves U/D/L/R: step 4 (big) / 2 (small); blocked if a target cell is not floor; cost 1 [a1..a8].
* Diamond (shrink/grow) cost 5 = 3 selects + completing click 2; column shot cost 4 = 2 + 2 [unique cost fit, work/fit8.py].
  Shrink keeps top-left [a1 tr17, a2 tr111, a4 tr141]. Grow: hypothesis `hgp` (below).
* Shot: may open the barrier only from columns allowed by `mayOpen` (hypothesis `hopen`).
* Pad (cells (34..35, 23..24)): the first overlap by any block, after any action, gives net -10
  [a2 tr103 small, a7 tr214 big partial, a8 tr242 big full; refund form "subtract 10" fixed by a4 tr142].
  Over-generous: a2 tr98 (big block moving to cover it fully) did NOT trigger; the model lets it.
* Budget: an action that would take `used` above B is GAME_OVER even if it reaches the exit [a8 tr254]; B = 18 [unique fit].
* Exact pattern matching (one click completing two patterns fires nothing) [a6]: `superset = false`.
* Not modelled (all make the model looser): facing (shot direction, turning on blocked moves), the row a shot is fired
  from, the order of clicks (macro spells), clicks on world/cards (free no-ops: `free`, cost 0).
Proof method: a Bellman cost-to-go table `V`, computed outside Lean (work/gen_final.py), is CHECKED inside Lean by the

`used + V` never decreases along a run, and V start = 19 > B.

structure St where
  big : Bool
  op : Bool
  pad : Bool
  r : Nat
  c : Nat
  deriving Repr, DecidableEq

inductive Act where
  | U | D | L | R | dia | shot | combo | free
  deriving Repr, DecidableEq

def floorCell (r c : Nat) (op : Bool) : Bool :=
  (decide (21 ≤ r ∧ r ≤ 36 ∧ 21 ≤ c ∧ c ≤ 44)) ||
  (decide (9 ≤ r ∧ r ≤ 20 ∧ 31 ≤ c ∧ c ≤ 32)) ||
  (decide (29 ≤ r ∧ r ≤ 32 ∧ 48 ≤ c ∧ c ≤ 56)) ||
  (decide (28 ≤ r ∧ r ≤ 33 ∧ 57 ≤ c ∧ c ≤ 61)) ||
  (op && decide (29 ≤ r ∧ r ≤ 32 ∧ 45 ≤ c ∧ c ≤ 47)) ||
  -- attempt 3 (tr 127): the destroyed switch box becomes floor
  (op && decide (5 ≤ r ∧ r ≤ 8 ∧ 30 ≤ c ∧ c ≤ 33))

def sz (big : Bool) : Nat := if big then 4 else 2

def fits (big op : Bool) (r c : Nat) : Bool :=
  (List.range (sz big)).all fun i => (List.range (sz big)).all fun j => floorCell (r + i) (c + j) op

/-- Attempt 7 amendment: a BIG block partly covering the pad (after a grow) also triggered the refund.
    The model now triggers on ANY overlap by a block of either size (looser than the game:
    a2 tr 98, a big block moving to cover it fully, did not trigger). -/
def onPad (big : Bool) (r c : Nat) : Bool :=
  (List.range (sz big)).any fun i => (List.range (sz big)).any fun j =>
    decide ((r + i = 34 ∨ r + i = 35) ∧ (c + j = 23 ∨ c + j = 24))

def atExit (big : Bool) (r c : Nat) : Bool :=
  (List.range (sz big)).any fun i => (List.range (sz big)).any fun j =>
    decide (28 ≤ r + i ∧ r + i ≤ 33 ∧ 57 ≤ c + j ∧ c + j ≤ 61)

def valid (s : St) : Bool :=
  decide (s.r % 2 = 1 ∧ 5 ≤ s.r ∧ s.r ≤ 35 ∧ s.c % 2 = 1 ∧ 21 ≤ s.c ∧ s.c ≤ 61) && fits s.big s.op s.r s.c

def goal (s : St) : Bool := s.op && atExit s.big s.r s.c

/-- pad bookkeeping after an action of gross cost `k`: returns (state, net cost). -/
def post (s : St) (k : Int) : St × Int :=
  if !s.pad && onPad s.big s.r s.c then ({ s with pad := true }, k - 10) else (s, k)

def mv (s : St) (dr dc : Int) : St :=
  let st : Int := if s.big then 4 else 2
  let nr : Int := s.r + dr * st
  let nc : Int := s.c + dc * st
  if 0 ≤ nr ∧ 0 ≤ nc ∧ fits s.big s.op nr.toNat nc.toNat = true then { s with r := nr.toNat, c := nc.toNat } else s

def bigAt (s : St) (r c : Nat) : St := { s with big := true, r := r, c := c }

/-- Attempt 6 amendment (tr: grow with (r-2,c-2) blocked landed at (r, c-2)).
    Possible grow outcomes of a SMALL block: observed preference (r-2,c-2), then (r,c-2);
    if both are blocked (never observed) every other fitting placement, or staying small, is allowed. -/
def growOuts (s : St) : List St :=
  if 2 ≤ s.r ∧ 2 ≤ s.c ∧ fits true s.op (s.r - 2) (s.c - 2) = true then [bigAt s (s.r - 2) (s.c - 2)]
  else if 2 ≤ s.c ∧ fits true s.op s.r (s.c - 2) = true then [bigAt s s.r (s.c - 2)]
  else (if 2 ≤ s.r ∧ fits true s.op (s.r - 2) s.c = true then [bigAt s (s.r - 2) s.c] else []) ++
       (if fits true s.op s.r s.c = true then [bigAt s s.r s.c] else []) ++ [s]

/-- diamond: shrink keeps top-left; grow is resolved by the engine's choice `gp` (hypothesis `hgp`). -/
def toggleWith (gp : St → St) (s : St) : St :=
  if s.big then { s with big := false } else gp s

/-- one action: (next state, net cost). `opens` decides whether a shot opens the barrier. -/
def step (opens : St → Bool) (superset : Bool) (gp : St → St) (s : St) : Act → St × Int
  | .U => post (mv s (-1) 0) 1
  | .D => post (mv s 1 0) 1
  | .L => post (mv s 0 (-1)) 1
  | .R => post (mv s 0 1) 1
  | .dia => post (toggleWith gp s) 5
  | .shot => post { s with op := s.op || opens s } 4
  | .combo => if superset then post (let t := toggleWith gp s; { t with op := t.op || opens t }) 6 else (s, 1)
  | .free => (s, 0)

/-- Budgeted run: an action whose net cost would push `used` above `B` ends the game (GAME_OVER).
    (Checking AFTER the refund is the permissive choice.) -/
def winsWithin (opens : St → Bool) (superset : Bool) (gp : St → St) (B : Int) : St → Int → List Act → Bool
  | s, _, [] => goal s
  | s, u, a :: rest => goal s ||
      (let p := step opens superset gp s a
       if u + p.2 > B then false else winsWithin opens superset gp B p.1 (u + p.2) rest)

def V (s : St) : Int := look tab (key s) 0

/-- Columns whose shot is NOT excluded by the evidence. Observed: col 31 opens (a3 tr 127, a4 tr 154);
    big@29 fails (a1 tr 76,79), big@33 and small@33 fail (a5). A shot travels straight in the facing
    direction (a1 tr 83 frames, a3 tr 127 frames), so only blocks overlapping the box columns 30-33 can
    reach the box: big c ∈ {27,29,31,33}, small c ∈ {29,31,33}. Untested ones are LEFT OPEN: big@27, small@29. -/
def mayOpen (s : St) : Bool :=
  decide (s.c = 31) || (s.big && decide (s.c = 27)) || (!s.big && decide (s.c = 29))

/-- all possible outcomes of an action for ANY `opens` satisfying `hopen`, with superset = false. -/
def outs (s : St) (a : Act) : List (St × Int) :=
  match a with
  | .shot => if mayOpen s then [post s 4, post { s with op := true } 4] else [post s 4]
  | .dia => if s.big then [post { s with big := false } 5] else (growOuts s).map (fun t => post t 5)
  | _ => [step (fun _ => false) false (fun x => x) s a]

def edgeOk (s : St) (a : Act) : Bool :=
  (outs s a).all fun p => valid p.1 && decide (V s ≤ p.2 + V p.1)

def acts : List Act := [.U, .D, .L, .R, .dia, .shot, .combo, .free]

def stOf (b o p : Bool) (r c : Nat) : St := { big := b, op := o, pad := p, r := r, c := c }

/-- one slice of the state space (fixed size/opened/pad flags), checked separately to bound memory. -/
def chunk (P : St → Bool) (b o p : Bool) : Bool :=
  (List.range 16).all fun i => (List.range 21).all fun j =>
    !valid (stOf b o p (5 + 2 * i) (21 + 2 * j)) || P (stOf b o p (5 + 2 * i) (21 + 2 * j))

/-- "P holds on every valid lattice state", as a computable Bool. -/
def allValid (P : St → Bool) : Bool :=
  [false, true].all fun b => [false, true].all fun o => [false, true].all fun p => chunk P b o p

theorem allValid_spec (P : St → Bool) (h : allValid P = true) (s : St) (hv : valid s = true) :
    P s = true := by
  have hv0 := hv
  rcases s with ⟨b, o, p, r, c⟩
  simp only [valid, Bool.and_eq_true, decide_eq_true_eq] at hv
  obtain ⟨⟨_, hr2, hr3, _, hc2, hc3⟩, _⟩ := hv
  simp only [allValid, chunk, List.all_eq_true, List.mem_range] at h
  have hb : b ∈ [false, true] := by cases b <;> simp
  have ho : o ∈ [false, true] := by cases o <;> simp
  have hp : p ∈ [false, true] := by cases p <;> simp
  have h1 := h b hb o ho p hp ((r - 5) / 2) (by omega) ((c - 21) / 2) (by omega)
  have er : 5 + 2 * ((r - 5) / 2) = r := by omega
  have ec : 21 + 2 * ((c - 21) / 2) = c := by omega
  rw [er, ec] at h1
  simp only [stOf] at h1
  rw [hv0] at h1
  simpa using h1

theorem edge (s : St) (hv : valid s = true) (a : Act) : edgeOk s a = true := by
  have h := allValid_spec _ checkAll_true s hv
  simp only [List.all_eq_true] at h
  exact h a (by cases a <;> simp [acts])

theorem goal_le (s : St) (hv : valid s = true) (hg : goal s = true) : V s ≤ 0 := by
  have h := allValid_spec _ goalCheck_true s hv
  simp [hg] at h
  exact h

/-- the real step lands in the enumerated outcome list, for every `opens` obeying `hopen`
    and every grow choice `gp` obeying `hgp`. -/
theorem step_mem (opens : St → Bool) (hopen : ∀ s, opens s = true → mayOpen s = true)
    (gp : St → St) (hgp : ∀ s, s.big = false → gp s ∈ growOuts s)
    (s : St) (a : Act) : step opens false gp s a ∈ outs s a := by
  cases a with
  | shot =>
    simp only [step, outs]
    cases hm : mayOpen s
    · have hf : opens s = false := by
        cases ho : opens s
        · rfl
        · have := hopen s ho; rw [hm] at this; exact absurd this (by simp)
      simp [hf]
    · cases ho : opens s <;> simp
  | dia =>
    simp only [step, outs, toggleWith]
    cases hb : s.big
    · simp only [Bool.false_eq_true, if_false]
      exact List.mem_map_of_mem (hgp s hb)
    · simp
  | _ => simp [step, outs]

theorem inv (opens : St → Bool) (hopen : ∀ s, opens s = true → mayOpen s = true)
    (gp : St → St) (hgp : ∀ s, s.big = false → gp s ∈ growOuts s) (B : Int) :
    ∀ (as : List Act) (s : St) (u : Int), valid s = true → u ≤ B →
      winsWithin opens false gp B s u as = true → u + V s ≤ B := by
  intro as
  induction as with
  | nil =>
    intro s u hv hu hw
    simp only [winsWithin] at hw
    have := goal_le s hv hw
    omega
  | cons a rest ih =>
    intro s u hv hu hw
    simp only [winsWithin, Bool.or_eq_true] at hw
    rcases hw with hg | hw
    · have := goal_le s hv hg; omega
    · have hmem := step_mem opens hopen gp hgp s a
      have he := edge s hv a
      simp only [edgeOk, List.all_eq_true, Bool.and_eq_true, decide_eq_true_eq] at he
      obtain ⟨hv', hle⟩ := he _ hmem
      split at hw
      · exact absurd hw (by simp)
      · next hb =>
        have := ih _ _ hv' (by omega) hw
        omega

/-- **MAIN THEOREM.** For every budget `0 ≤ B ≤ 18` (measured level-4 budget: 18, unique cost fit over
    all eight attempts), for every way the engine resolves a shot (`opens`) consistent with the evidence
    (`hopen`), and for every way it places a grow (`gp`) consistent with the evidence (`hgp`),
    NO action list wins level 4 from the start position.  Exact pattern matching (superset = false)
    is verified (a6) and fixed. -/
theorem no_win_within_budget
    (opens : St → Bool)
    -- hopen: a shot can open the barrier only from a column not excluded by observation (see `mayOpen`).
    (hopen : ∀ s, opens s = true → mayOpen s = true)
    (gp : St → St)
    -- hgp: grow lands at (r-2,c-2) if it fits [a2 tr107, a3 tr124, a4 tr148], else (r,c-2) if it fits
    -- [a6], else (never observed) any other fitting placement or no change.
    (hgp : ∀ s, s.big = false → gp s ∈ growOuts s)
    (B : Int) (hB0 : 0 ≤ B) (hB : B ≤ 18) :
    ∀ as : List Act, winsWithin opens false gp B start 0 as = false := by
  intro as
  cases hw : winsWithin opens false gp B start 0 as
  · rfl
  · have := inv opens hopen gp hgp B as start 0 start_valid hB0 hw
    rw [V_start] at this
    omega

/-- the measured budget instance -/
theorem no_win_at_measured_budget
    (opens : St → Bool) (hopen : ∀ s, opens s = true → mayOpen s = true)
    (gp : St → St) (hgp : ∀ s, s.big = false → gp s ∈ growOuts s) :
    ∀ as : List Act, winsWithin opens false gp L4budget start 0 as = false :=
  no_win_within_budget opens hopen gp hgp L4budget (by decide) (by decide)

/-! ## non-vacuity: the least winning budget is 19 -/
def opens31 : St → Bool := fun s => decide (s.c = 31)
theorem opens31_ok : ∀ s, opens31 s = true → mayOpen s = true := by
  intro s h; simp [opens31] at h; simp [mayOpen, h]

/-- the observed grow rule: first entry of `growOuts`. -/
def gpReal (s : St) : St := (growOuts s).headD s
theorem gpReal_ok : ∀ s, s.big = false → gpReal s ∈ growOuts s := by
  intro s _
  unfold gpReal growOuts
  split
  · simp
  · split
    · simp
    · split <;> split <;> simp

/-- exactly the route played in attempt 8 (tr 229-254): it faces UP before the shot, as the game requires. -/
def witness : List Act := [.L, .L, .L, .D, .D, .dia, .D, .dia, .R, .R, .U, .shot, .R, .R, .R, .R, .R, .R]

/-- **WITNESS THEOREM.** With budget 19 there is a winning action list (for admissible `opens31`, `gpReal`). -/
theorem win_at_19 : ∃ as : List Act, winsWithin opens31 false gpReal 19 start 0 as = true :=
  ⟨witness, by decide⟩

/-- Least budget: 19 wins (for an admissible instance), every budget ≤ 18 loses (for all admissible ones). -/
theorem least_winning_budget_is_19 :
    (∃ as : List Act, winsWithin opens31 false gpReal 19 start 0 as = true) ∧
    (∀ B : Int, 0 ≤ B → B ≤ 18 → ∀ as : List Act, winsWithin opens31 false gpReal B start 0 as = false) :=
  ⟨win_at_19, fun B hB0 hB => no_win_within_budget opens31 opens31_ok gpReal gpReal_ok B hB0 hB⟩
