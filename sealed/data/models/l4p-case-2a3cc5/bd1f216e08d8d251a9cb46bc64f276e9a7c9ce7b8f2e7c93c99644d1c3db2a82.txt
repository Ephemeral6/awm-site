# Rewrites the table part of lean/L4ModelV3Tab.lean: the 1489-entry least-cost table becomes a
# binary search tree `tree : T`; the kernel check runs over the tree's entries only (not 131072 keys).
import re, sys
ents = []
for line in open('work/tab4.lean.txt', encoding='utf-8'):
    m = re.match(r'\s*\((\d+), (\d+)\)', line)
    if m:
        ents.append((int(m.group(1)), int(m.group(2))))
ents.sort()

def tree(lo, hi):
    if lo >= hi:
        return '.leaf'
    mid = (lo + hi) // 2
    k, v = ents[mid]
    return '(.node %s %d %d %s)' % (tree(lo, mid), k, v, tree(mid + 1, hi))

head = '''/-- binary search tree of (key, least spent budget); generated by work/table4.py + work/gen_tab.py -/
inductive T | leaf | node (l : T) (k v : Nat) (r : T)

def T.find : T → Nat → Option Nat
  | .leaf, _ => none
  | .node l k v r, x => if x < k then l.find x else if x = k then some v else r.find x

def T.toList : T → List (Nat × Nat)
  | .leaf => []
  | .node l k v r => l.toList ++ (k, v) :: r.toList

theorem T.find_mem : ∀ (t : T) (x v : Nat), t.find x = some v → (x, v) ∈ t.toList
  | .leaf, x, v, h => by simp [T.find] at h
  | .node l k w r, x, v, h => by
    simp only [T.find] at h
    simp only [T.toList, List.mem_append, List.mem_cons]
    split at h
    · exact Or.inl (T.find_mem l x v h)
    · split at h
      · rename_i hx
        simp only [Option.some.injEq] at h
        subst hx; subst h
        exact Or.inr (Or.inl rfl)
      · exact Or.inr (Or.inr (T.find_mem r x v h))

def tree : T :=
  %s

def gk (k : Nat) : Option Nat := tree.find k

''' % tree(0, len(ents))

tail = '''def checkAll : Bool := tree.toList.all fun e => allActs.all (ok (decode e.1) e.2)

theorem checkAll_true : checkAll = true := by decide +kernel

/-- every state in the table passes the transition check. -/
theorem gk_ok (k v : Nat) (h : gk k = some v) : allActs.all (ok (decode k) v) = true := by
  have hc := checkAll_true
  simp only [checkAll, List.all_eq_true] at hc
  exact hc (k, v) (T.find_mem tree k v h)

'''

p = 'lean/L4ModelV3Tab.lean'
s = open(p, encoding='utf-8').read()
i = s.index('/-- the reachable-state table')
j = s.index('def inB')
s = s[:i] + head + s[j:]
i = s.index('def chk (k : Nat)')
j = s.index('end L4.V3')
s = s[:i] + tail + s[j:]
open(p, 'w', encoding='utf-8').write(s)
print('entries', len(ents))
