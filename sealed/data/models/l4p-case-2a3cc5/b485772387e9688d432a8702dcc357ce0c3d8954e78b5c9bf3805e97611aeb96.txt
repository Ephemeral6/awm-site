namespace L4.V4
open L4 (Dir shift inSlot box)

structure P where
  r : Nat
  c : Nat
  n : Nat
  face : Dir
  op : Bool
  rf : Bool
  deriving DecidableEq, Repr

def inFp (r c n y x : Nat) : Bool := Nat.ble r y && Nat.blt y (r + n) && Nat.ble c x && Nat.blt x (c + n)

/-- (y,x) lies in the 9 half (the half on the facing side). -/
def inNine (r c n : Nat) (d : Dir) (y x : Nat) : Bool :=
  inFp r c n y x &&
  match d with
  | .U => Nat.blt y (r + n / 2)
  | .D => Nat.ble (r + n - n / 2) y
  | .L => Nat.blt x (c + n / 2)
  | .R => Nat.ble (c + n - n / 2) x

def markerCells : List (Nat × Nat) := [(34, 23), (34, 24), (35, 23), (35, 24)]

def straddle (r c n : Nat) (d : Dir) : Bool :=
  markerCells.any (fun q => inNine r c n d q.1 q.2) &&
  markerCells.any (fun q => inFp r c n q.1 q.2 && !inNine r c n d q.1 q.2)

def slideT (op : Bool) (n : Nat) (d : Dir) (p : Nat × Nat) : Nat → Nat → Bool → Nat × Bool
  | 0, k, t => (k, t)
  | m + 1, k, t =>
    let q := shift d (k + 1) p
    if L4.V2.fpAll (L4.V2.floorC op) n q.1 q.2 then
      slideT op n d p m (k + 1) (t || straddle q.1 q.2 n d)
    else (k, t)

inductive Kind | norm | refill

inductive Res | win | to (t : P) (kd : Kind)

inductive Act | mv (d : Dir) | F (i : Nat) | M6

def cost : Act → Nat
  | .mv _ => 1
  | .F _ => 5
  | .M6 => 4

def fits (op : Bool) (r c ay ax : Nat) : Bool :=
  Nat.ble ay r && Nat.ble ax c && L4.V2.fpAll (L4.V2.floorC op) 4 (r - ay) (c - ax)

def absStep (p : P) : Act → Res
  | .mv d =>
    let q := shift d p.n (p.r, p.c)
    if L4.V2.fpAny inSlot p.n q.1 q.2 then .win else
    let st := slideT p.op p.n d (p.r, p.c) p.n 0 (straddle p.r p.c p.n d)
    let t := shift d st.1 (p.r, p.c)
    if !p.rf && st.2 then .to ⟨t.1, t.2, p.n, d, p.op, true⟩ .refill
    else .to ⟨t.1, t.2, p.n, d, p.op, p.rf⟩ .norm
  | .F i =>
    if p.n == 4 then .to { p with n := 2 } .norm else
    if fits p.op p.r p.c 2 2 then .to { p with r := p.r - 2, c := p.c - 2, n := 4 } .norm else
    let a := L4.V3.anch i
    if fits p.op p.r p.c a.1 a.2
    then .to { p with r := p.r - a.1, c := p.c - a.2, n := 4 } .norm else .to p .norm
  | .M6 => .to { p with op := p.op || p.c == 31 } .norm

structure St where
  p : P
  k : Nat
  won : Bool

def step (B : Nat) (s : St) (a : Act) : St :=
  if s.won = false ∧ s.k + cost a ≤ B then
    match absStep s.p a with
    | .win => ⟨s.p, s.k, true⟩
    | .to t .norm => ⟨t, s.k + cost a, false⟩
    | .to t .refill => ⟨t, s.k + 1 - 10, false⟩
  else s

def run (B : Nat) (s : St) (as : List Act) : St := as.foldl (step B) s

def startP : P := ⟨25, 37, 4, .D, false, false⟩
def start : St := ⟨startP, 0, false⟩

def fi : Dir → Nat
  | .U => 0 | .D => 1 | .L => 2 | .R => 3

def toDir (i : Nat) : Dir := if i = 0 then .U else if i = 1 then .D else if i = 2 then .L else .R

def key (p : P) : Nat :=
  ((p.r * 64 + p.c) * 8 + p.n) * 16 + fi p.face * 4 + (if p.op then 2 else 0) + (if p.rf then 1 else 0)

def decode (k : Nat) : P :=
  ⟨k / 8192, k / 128 % 64, k / 16 % 8, toDir (k / 4 % 4), k / 2 % 2 == 1, k % 2 == 1⟩

/-- binary search tree of (key, least spent budget); generated by work/table6.py + work/gen_v4.py -/
inductive T | leaf | node (l : T) (k v : Nat) (r : T)

def T.find : T → Nat → Option Nat
  | .leaf, _ => none
  | .node l k v r, x => if x < k then l.find x else if x = k then some v else r.find x

def T.toList : T → List (Nat × Nat)
  | .leaf => []
  | .node l k v r => l.toList ++ (k, v) :: r.toList

theorem T.find_mem : ∀ (t : T) (x v : Nat), t.find x = some v → (x, v) ∈ t.toList
  | .leaf, x, v, h => by simp [T.find] at h
  | .node l k w r, x, v, h => by
    simp only [T.find] at h
    simp only [T.toList, List.mem_append, List.mem_cons]
    split at h
    · exact Or.inl (T.find_mem l x v h)
    · split at h
      · rename_i hx
        simp only [Option.some.injEq] at h
        subst hx; subst h
        exact Or.inr (Or.inl rfl)
      · exact Or.inr (Or.inr (T.find_mem r x v h))
