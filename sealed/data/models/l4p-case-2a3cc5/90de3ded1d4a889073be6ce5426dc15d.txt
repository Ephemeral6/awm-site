def ok (p : P) (g : Nat) (a : Act) : Bool :=
  match absStep p a with
  | .win => decide (18 < g + 1)
  | .to t .norm => decide (18 < g + cost a) ||
      (inB t && match gk (key t) with
                | some v => decide (v ≤ g + cost a)
                | none => false)
  | .to t .refill => decide (18 < g + 1) ||
      (inB t && match gk (key t) with
                | some v => decide (v ≤ min (g + 1 - 10) 5)
                | none => false)
