def marker (y x : Nat) : Bool := box 34 35 23 24 y x

structure P where
  r : Nat
  c : Nat
  n : Nat
  op : Bool
  rf : Bool
  deriving DecidableEq, Repr

def slideT (op : Bool) (n : Nat) (d : Dir) (p : Nat × Nat) : Nat → Nat → Bool → Nat × Bool
  | 0, k, t => (k, t)
  | m + 1, k, t =>
    let q := shift d (k + 1) p
    if L4.V2.fpAll (L4.V2.floorC op) n q.1 q.2 then
      slideT op n d p m (k + 1) (t || L4.V2.fpAny marker n q.1 q.2)
    else (k, t)

inductive Kind | norm | refill

inductive Res | win | to (t : P) (kd : Kind)

/-- grow anchors (offsets subtracted from the top-left); index ≥ 4 all mean (0,2). -/
def anch : Nat → Nat × Nat
  | 0 => (2, 2) | 1 => (0, 0) | 2 => (1, 1) | 3 => (2, 0) | _ => (0, 2)

inductive Act | mv (d : Dir) | F (i : Nat) | M6

def cost : Act → Nat
  | .mv _ => 1
  | .F _ => 5
  | .M6 => 4

def absStep (p : P) : Act → Res
  | .mv d =>
    let q := shift d p.n (p.r, p.c)
    if L4.V2.fpAny inSlot p.n q.1 q.2 then .win else
    let st := slideT p.op p.n d (p.r, p.c) p.n 0 false
    let t := shift d st.1 (p.r, p.c)
    if p.n == 2 && !p.rf && st.2 then .to ⟨t.1, t.2, p.n, p.op, true⟩ .refill
    else .to ⟨t.1, t.2, p.n, p.op, p.rf⟩ .norm
  | .F i =>
    if p.n == 4 then .to ⟨p.r, p.c, 2, p.op, p.rf⟩ .norm else
    let a := anch i
    if Nat.ble a.1 p.r && Nat.ble a.2 p.c && L4.V2.fpAll (L4.V2.floorC p.op) 4 (p.r - a.1) (p.c - a.2)
    then .to ⟨p.r - a.1, p.c - a.2, 4, p.op, p.rf⟩ .norm else .to p .norm
  | .M6 => .to ⟨p.r, p.c, p.n, p.op || p.c == 31, p.rf⟩ .norm

structure St where
  p : P
  k : Nat
  won : Bool

def step (B : Nat) (s : St) (a : Act) : St :=
  if s.won = false ∧ s.k + cost a ≤ B then
    match absStep s.p a with
    | .win => ⟨s.p, s.k, true⟩
    | .to t .norm => ⟨t, s.k + cost a, false⟩
    | .to t .refill => ⟨t, min (s.k + 1 - 10) 5, false⟩
  else s

def run (B : Nat) (s : St) (as : List Act) : St := as.foldl (step B) s

def decode (k : Nat) : P := ⟨k / 2048, k / 32 % 64, k / 4 % 8, k / 2 % 2 == 1, k % 2 == 1⟩

def T.find : T → Nat → Option Nat
  | .leaf, _ => none
  | .node l k v r, x => if x < k then l.find x else if x = k then some v else r.find x

def T.toList : T → List (Nat × Nat)
  | .leaf => []
  | .node l k v r => l.toList ++ (k, v) :: r.toList

theorem T.find_mem : ∀ (t : T) (x v : Nat), t.find x = some v → (x, v) ∈ t.toList
  | .leaf, x, v, h => by simp [T.find] at h
  | .node l k w r, x, v, h => by
    simp only [T.find] at h
    simp only [T.toList, List.mem_append, List.mem_cons]
    split at h
    · exact Or.inl (T.find_mem l x v h)
    · split at h
      · rename_i hx
        simp only [Option.some.injEq] at h
        subst hx; subst h
        exact Or.inr (Or.inl rfl)
      · exact Or.inr (Or.inr (T.find_mem r x v h))

def gk (k : Nat) : Option Nat := tree.find k

def inB (t : P) : Bool := decide (t.r < 64) && decide (t.c < 64) && decide (t.n < 8)

def ok (p : P) (g : Nat) (a : Act) : Bool :=
  match absStep p a with
  | .win => decide (18 < g + 1)
  | .to t .norm => decide (18 < g + cost a) ||
      (inB t && match gk (key t) with
                | some v => decide (v ≤ g + cost a)
                | none => false)
  | .to t .refill => decide (18 < g + 1) ||
      (inB t && match gk (key t) with
                | some v => decide (v ≤ min (g + 1 - 10) 5)
                | none => false)

def allActs : List Act :=
  [.mv .U, .mv .D, .mv .L, .mv .R, .F 0, .F 1, .F 2, .F 3, .F 4, .M6]

def checkAll : Bool := tree.toList.all fun e => allActs.all (ok (decode e.1) e.2)

theorem checkAll_true : checkAll = true := by decide +kernel

/-- every state in the table passes the transition check. -/
theorem gk_ok (k v : Nat) (h : gk k = some v) : allActs.all (ok (decode k) v) = true := by
  have hc := checkAll_true
  simp only [checkAll, List.all_eq_true] at hc
  exact hc (k, v) (T.find_mem tree k v h)
