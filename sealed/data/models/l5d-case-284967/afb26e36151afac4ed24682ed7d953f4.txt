def inRect (r c r0 r1 c0 c1 : Int) : Bool := r0 ≤ r && r ≤ r1 && c0 ≤ c && c ≤ c1

def ccBar (r c : Int) : Bool := inRect r c 33 36 29 31
def ddBar (r c : Int) : Bool := inRect r c 11 13 33 36
def ccBox (r c : Int) : Bool := inRect r c 9 12 20 23
def ddBox (r c : Int) : Bool := inRect r c 14 17 55 58
def goalIn (r c : Int) : Bool := inRect r c 1 4 33 36
def padCell (r c : Int) : Bool := inRect r c 37 38 49 50

def staticFloor (r c : Int) : Bool :=
  if r < 0 || c < 0 || r > 41 || c > 63 then false
  else Nat.testBit (floorMask.getD r.toNat 0) c.toNat

structure St where
  r : Int
  c : Int
  s : Int
  f : Nat      -- 0 up 1 down 2 left 3 right
  cc : Bool
  dd : Bool
  pad : Bool
  bud : Nat
  won : Bool
deriving Repr, DecidableEq

def floorAt (st : St) (r c : Int) : Bool :=
  staticFloor r c || (st.cc && (ccBar r c || ccBox r c)) || (st.dd && (ddBar r c || ddBox r c))

def blockAll (p : Int → Int → Bool) (r c s : Int) : Bool :=
  (List.range s.toNat).all fun i => (List.range s.toNat).all fun j => p (r + i) (c + j)

def blockAny (p : Int → Int → Bool) (r c s : Int) : Bool :=
  (List.range s.toNat).any fun i => (List.range s.toNat).any fun j => p (r + i) (c + j)

def dvec : Nat → Int × Int
  | 0 => (-1, 0) | 1 => (1, 0) | 2 => (0, -1) | _ => (0, 1)

/-- largest k ≤ s such that shifts 1..k are all floor (sliding). -/
def slide (st : St) (d : Nat) : Int :=
  let (dr, dc) := dvec d
  let ok : Nat → Bool := fun k => (List.range k).all fun j =>
    blockAll (floorAt st) (st.r + dr * (j + 1)) (st.c + dc * (j + 1)) st.s
  ((List.range (st.s.toNat + 1)).filter ok).foldl (fun (a : Int) (k : Nat) => max a (k : Int)) 0

inductive Act where
  | mv (d : Nat) | shrink | cast6 | tele
deriving Repr, DecidableEq

def cost : Act → Nat
  | .mv _ => 1 | .shrink => 4 | .cast6 => 3 | .tele => 3

/-- probe: top-left 2×2 slid in facing direction; returns (hit cc box, hit dd box). -/
def probe (st : St) : Bool × Bool :=
  let (dr, dc) := dvec st.f
  let rec go : Nat → Int → Int → Bool × Bool
    | 0, _, _ => (false, false)
    | n + 1, r, c =>
      let r' := r + dr; let c' := c + dc
      if blockAny ccBox r' c' 2 && !st.cc then (true, false)
      else if blockAny ddBox r' c' 2 && !st.dd then (false, true)
      else if blockAll (floorAt st) r' c' 2 then go n r' c'
      else (false, false)
  go 64 st.r st.c

def apply (st : St) : Act → St
  | .mv d =>
    let k := slide st d
    let (dr, dc) := dvec d
    let st1 := { st with f := d, r := st.r + dr * k, c := st.c + dc * k }
    let st2 := if !st1.pad && blockAny padCell st1.r st1.c st1.s
               then { st1 with pad := true, bud := st1.bud + 8 } else st1
    if blockAny goalIn st2.r st2.c st2.s then { st2 with won := true } else st2
  | .shrink => if st.s = 4 then { st with s := 2 } else st
  | .cast6 =>
    let (a, b) := probe st
    { st with cc := st.cc || a, dd := st.dd || b }
  | .tele => if st.s = 2 then { st with r := 18, c := 34 } else { st with r := 17, c := 33 }

/-- R-budget: refused action ⇒ `none` (game over). -/
def step (st : St) (a : Act) : Option St :=
  if st.won then some st
  else if cost a ≤ st.bud then some (apply { st with bud := st.bud - cost a } a) else none

def run : St → List Act → Option St
  | st, [] => some st
  | st, a :: as => match step st a with
    | none => none
    | some st' => run st' as

def start (b : Nat) : St :=
  { r := 29, c := 17, s := 4, f := 1, cc := false, dd := false, pad := false, bud := b, won := false }

def wins (b : Nat) (as : List Act) : Bool :=
  match run (start b) as with
  | some st => st.won
  | none => false
