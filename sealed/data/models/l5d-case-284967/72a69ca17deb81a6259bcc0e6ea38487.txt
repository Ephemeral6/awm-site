N0=35
PAT={0b010101010:'f',0b010010010:'6',0b000010011:'b'}   # bit 3*r+c
def disp(rem): return 4*max(0,min(32,math.ceil(32*rem/35-1e-9)))
def step(st,act):
    """st=(r,c,s,f,cc,dd,pad,sel,rem); act=('mv',d)|('click',i)|('free',)
    returns (st', won) or None (game over)"""
    r,c,s,f,cc,dd,pad,sel,rem=st
    w=(r,c,s,f,cc,dd,pad)
    if act[0]=='free': return st,False
    if act[0]=='mv':
        if rem<1: return None
        n,_,g,won=nxt3(w,act[1]); return n+(sel,rem-1+g),won
    i=act[1]; s2=sel^(1<<i)
    if s2 in PAT:
        if rem<2: return None
        p=PAT[s2]; a={'f':(4 if s==4 else 7),'6':5,'b':6}[p]
        n,_,_,_=nxt3(w,a); return n+(0,rem-2),False
    if rem<1: return None
    return w+(s2,rem-1),False
def decode(a):
    if not a.startswith('ACTION6'): return ('mv',int(a[6])-1)
    x,y=map(int,a.split(':')[1].split(','))
    for rr in range(3):
        for cc_ in range(3):
            if abs(x-(25+5*cc_))<=1 and abs(y-(50+5*rr))<=1: return ('click',3*rr+cc_)
    return ('free',)
def sel(g): return sum(1<<(3*r+c) for r in range(3) for c in range(3) if g[50+5*r][25+5*c]==14)
