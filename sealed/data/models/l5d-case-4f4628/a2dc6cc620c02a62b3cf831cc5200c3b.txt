theorem look_chk (q : PS) (d : Int) (h : look q = some d) : chk (q, d) = true := by
  unfold look at h
  obtain ⟨t, ht, h'⟩ := CTree.lookup_mem _ _ _ _ h
  have hall := chunks_ok t ht
  exact List.all_eq_true.mp hall _ (Tree.lookup_mem _ _ _ _ h')

theorem chk_act (q : PS) (d : Int) (h : chk (q, d) = true) (a : Act) :
    (let r := pstep T4 T2 q a
     let bonus : Int := 90 * ((b2n r.1.pk : Int) - (b2n q.pk : Int))
     if r.2 then decide (d ≤ wt a - bonus)
     else match look r.1 with
       | some d' => decide (d ≤ wt a - bonus + d')
       | none => false) = true :=
  List.all_eq_true.mp h a (mem_allActs a)

theorem bound : ∀ (as : List Act) (q : PS) (d : Int) (N k : Nat) (p : Bool),
    look q = some d → stats T4 T2 q as = some (N, k, p) →
    d ≤ 9 * (N : Int) + 10 * (k : Int) - 90 * ((b2n p : Int) - (b2n q.pk : Int)) := by
  intro as
  induction as with
  | nil => intro q d N k p _ h; simp [stats] at h
  | cons a as ih =>
    intro q d N k p hl hs
    have hc := chk_act q d (look_chk q d hl) a
    simp only [stats] at hs
    cases hw : (pstep T4 T2 q a).2 with
    | true =>
      simp only [hw, ite_true] at hs hc
      simp only [Option.some.injEq, Prod.mk.injEq] at hs
      obtain ⟨rfl, rfl, rfl⟩ := hs
      have := of_decide_eq_true hc
      simp only [wt] at this
      omega
    | false =>
      simp only [hw] at hs hc
      simp only [Bool.false_eq_true, ite_false] at hs hc
      cases hst : stats T4 T2 (pstep T4 T2 q a).1 as with
      | none => simp [hst] at hs
      | some x =>
        obtain ⟨N', k', p'⟩ := x
        simp only [hst, Option.map_some, Option.some.injEq, Prod.mk.injEq] at hs
        obtain ⟨rfl, rfl, rfl⟩ := hs
        cases hlk : look (pstep T4 T2 q a).1 with
        | none => simp [hlk] at hc
        | some d' =>
          simp only [hlk] at hc
          have h1 := of_decide_eq_true hc
          have h2 := ih _ d' N' k' p' hlk hst
          simp only [wt] at h1
          push_cast
          omega

theorem L5_no_win (t4 t2 : Nat × Nat) (h4 : t4 = (17,33)) (h2 : t2 = (37,15))
    (as : List Act) (N k : Nat) (p : Bool)
    (hwin : stats t4 t2 s0 as = some (N, k, p))
    (hE : (9 * (N - 1)) / 10 + k ≤ 31 + 9 * b2n p) : False := by
  subst h4; subst h2
  have hb := bound as s0 338 N k p look_s0 hwin
  have hp : b2n s0.pk = 0 := rfl
  rw [hp] at hb
  cases p <;> simp [b2n] at hb hE <;> omega
