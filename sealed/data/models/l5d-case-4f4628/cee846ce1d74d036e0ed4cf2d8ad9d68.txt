inductive Tree

def Tree.lookup : Tree → Nat → PS → Option Int
  | .leaf, _, _ => none
  | .node l k s0 d r, key, s =>
    if key < k then l.lookup key s
    else if k < key then r.lookup key s
    else if s0 = s then some d else none

def Tree.toList : Tree → List (PS × Int)
  | .leaf => []
  | .node l _ s d r => l.toList ++ (s, d) :: r.toList

theorem Tree.lookup_mem : ∀ (t : Tree) (key : Nat) (s : PS) (d : Int),
    t.lookup key s = some d → (s, d) ∈ t.toList := by
  intro t
  induction t with
  | leaf => intro key s d h; simp [Tree.lookup] at h
  | node l k s0 d0 r ihl ihr =>
    intro key s d h
    simp only [Tree.lookup] at h
    split at h
    · exact List.mem_append_left _ (ihl _ _ _ h)
    · split at h
      · exact List.mem_append_right _ (List.mem_cons_of_mem _ (ihr _ _ _ h))
      · split at h
        · rename_i heq
          subst heq
          simp at h
          subst h
          simp [Tree.toList]
        · simp at h

def Dir.idx : Dir → Nat | .U => 0 | .D => 1 | .L => 2 | .R => 3
def b2n (b : Bool) : Nat := if b then 1 else 0

def enc (q : PS) : Nat :=

   .growAt .o1 .o2, .growAt .o2 .o0, .growAt .o2 .o1, .growAt .o2 .o2]

theorem mem_allActs (a : Act) : a ∈ allActs := by
  cases a with
  | mv d => cases d <;> simp [allActs]
  | growAt a b => cases a <;> cases b <;> simp [allActs]

/-- energy weight of a macro: 9 per click (0.9 unit) + 10 per completion (1 unit), in tenths. -/

def CTree.lookup : CTree → Nat → PS → Option Int
  | .leaf t, k, q => t.lookup k q

def CTree.leaves : CTree → List Tree
  | .leaf t => [t]
  | .node _ l r => l.leaves ++ r.leaves

theorem CTree.lookup_mem : ∀ (c : CTree) (k : Nat) (q : PS) (d : Int),
    c.lookup k q = some d → ∃ t ∈ c.leaves, t.lookup k q = some d := by
