/-- **Main theorem (self-audit corrected model).** In the level-5 model with the beam rule R-col',
    every action sequence that wins costs at least 37 units, more than the measured 35-unit budget.
    It quantifies over every action sequence (moves, plus, col, tele) from the level start; no hypotheses. -/
theorem l5_impossible : ∀ (as : List Act) (C : Int), firstWin s0 as = some C → 35 < C := by
  intro as C hw
  have := lower_bound tab tab_ok as s0 37 C tab_s0 hw
  omega

/-- **Relaxed-model theorem.** Relax the budget from 35 to 37 (by 2 units): the same model has a winning
    route, and 37 is the least budget at which it wins. -/
theorem least_budget_is_37 :
    (∀ (as : List Act) (C : Int), firstWin s0 as = some C → 37 ≤ C) ∧ firstWin s0 witness = some 37 :=
  ⟨fun as C hw => lower_bound tab tab_ok as s0 37 C tab_s0 hw, witness_wins⟩

end L5
